\chapter{Quotient Space and Simplicial Complex}

\section{Equivalence Relations and Equivalence Classes}

\begin{definition}
    \begin{itemize}
        \item (\textbf{Equivalence Relation}): The equivalence relation on a set  $X$ is a relation such that:
        \begin{itemize}
            \item (Reflexive): $x \sim x$ for all $x \in X$;
            \item (Symmetric): $x \sim y$ implies $y \sim x$;
            \item (Transitive): $x \sim y,y\sim z$ implies $x\sim z$.
        \end{itemize}
        \item (\textbf{Partition}): Let $X$ be a non-empty set. A partition $\mathcal{P} := \{ P_i \mid i \in I \}$ of $X$ is a collection of subsets such that:
        \begin{itemize}
            \item $P_i \subseteq X$ is non-empty;
            \item $P_i \cap P_j = \emptyset$ if $i\neq j$;
            \item $\bigcup_{i \in I} P_i = X$.
        \end{itemize}
        \item (\textbf{Equivalence Class}): Let $X$ be a set with equivalence relation $\sim$. The equivalence class of $x\in X$ is defined by:
        $$[x] := \{ y\in X \mid x \sim y \}.$$
        \item (\textbf{Quotient Space}): The collection of all equivalence classes is called the equivalence space: 
        $$X / \sim  \; := \{ [x] : x\in X \}.$$
    \end{itemize}
\end{definition}

\noindent\textbf{Remark:} If $\{ p_j \mid j \in J \}$ is a partition of $X$, we can define an equivalence relation by:
$$x \sim y \Leftrightarrow x,y \in p_j \;\; for \;\; some \;\; j\in J.$$

\noindent \textbf{Example} (Mobius Strip):\\
Let $X = [0,1] \times [0,1]$. We define $(x_1,y_1) \sim (x_2,y_2)$ if
\begin{itemize}
    \item $x_1 = x_2, y_1 =y_2$ with $x_i \in (0,1)$;
    \item $x_1 = 0, x_2 = 1$ and $y_1 = 1-y_2$;
    \item $x_1 = 1,x_2=0$ and $y_1=1-y_2$.
\end{itemize}
Then the partition $X = \{ (x,y) \}_{x \in (0,1), y\in [0,1]} \cup \{ (1,y), (0,1-y) \}_{y\in [0,1]}$.

\section{Quotient Topology}

\begin{definition}{Quotient Topology}\\
Suppose $(X,\mathcal{T})$ is a topological space, and that $\sim$ is an equivalence relation on $X$. Define the \textit{canonical projection map} by 
$$p: X\rightarrow X/\sim \;\; with \;\; x \mapsto [x].$$
Define a family of subsets $\widetilde{\mathcal{T}}$ on $X/\sim$ by:
$$\widetilde{U} \subseteq X/\sim \;\; is \;\; in \;\; \widetilde{\mathcal{T}} \;\; if \;\; p^{-1} (\widetilde{U}) \in \mathcal{T}.$$
Then $\widetilde{\mathcal{T}}$ is a topology for $X/\sim$ and we say $(X/\sim,\widetilde{\mathcal{T}})$ the quotient topology of the quotient space.
\end{definition}

\noindent\textbf{Verification:}
\begin{itemize}
    \item Since $p^{-1}(X/\sim) = X \in \mathcal{T}$ and $p^{-1} ( \emptyset) =\emptyset \in \mathcal{T}$, then $X/\sim ,\emptyset \in \widetilde{\mathcal{T}}$.
    \item Suppose that $\widetilde{U} , \widetilde{V}\in \widetilde{\mathcal{T}}$, i.e.: $p^{-1}(\widetilde{U}), p^{-1}(V) \in \mathcal{T}$. 
    Then $p^{-1} (\widetilde{U} \cap \widetilde{V}) = p^{-1} ( \emptyset) \;\; or \;\; p^{-1}(\widetilde{U}) \in \mathcal{T}$. 
    Hence, $\widetilde{U} \cap \widetilde{V} \in \widetilde{\mathcal{T}}$.
    \item Suppose that $\widetilde{U}_{i} \in \widetilde{\mathcal{T}}$ $\forall i \in I$, i.e.: $p^{-1}(\widetilde{U}_i) \in \mathcal{T}$ for all $i\in I$.
    Hence, 
    $$p^{-1} (\bigcup _{i\in I} \widetilde{U}_i) = \bigcup_{i\in I} p^{-1} (\widetilde{U}_i) \in \mathcal{T} \Rightarrow \bigcup _{i\in I} \widetilde{U}_i \in \widetilde{\mathcal{T}}.$$
\end{itemize}

\begin{definition}{Induced Map}\\
Let $f: X \to Y$ be a map such that $f(x)=f(y)$ whenever $x \sim y$. Then there exists a well-defined unique map $g: X/\sim \to Y$, s.t.:
$$g([x]) := f(x)$$
i.e.: $f= g \circ p$ where $p$ denotes the canonical projection map. We refer to this map $g$ as the induced map, often denoted by $g= \overline{f}$.
\end{definition}

\begin{definition}{Open Map}\\
A function $f : (X,\mathcal{T}_X) \rightarrow (Y,\mathcal{T}_Y)$ is an open map if for each open $U \in \mathcal{T}_X$, $f(U) \in \mathcal{T}_Y$.
\end{definition}

\begin{proposition}
    $\widetilde{V} \subseteq X/\sim$ is closed $\Leftrightarrow$ $p^{-1} (\widetilde{V})$ is closed in $X$.
\end{proposition}

\begin{proof}
    $$p^{-1} ((X/\sim) - \widetilde{V}) = X - p^{-1} (\widetilde{V})$$
\end{proof}

\begin{proposition}
    Let $X,Z$ be two topological spaces, and let $\sim$ be an equivalence relation on $X$. Let $g : X/\sim \;\rightarrow Z$ be a function and let $p: X\rightarrow X/\sim$ be a projection map. Then:
    $$g \;\; is \;\; cont \Leftrightarrow g\circ p : X\rightarrow Z \;\; is \;\; cont.$$
\end{proposition}

\begin{proof}
    \begin{itemize}
        \item ($\Rightarrow$): $g,p$ cont $\Rightarrow$ $g\circ p$ cont.
        \item ($\Leftarrow$): For any open $U \subseteq Z$, $(g\circ p)^{-1} (U) = p^{-1}g^{-1} (U)$ is open in $X$. Then by the def of quotient topology, $g^{-1} (U)$ is open in $X/\sim$. Hence, $g$ is cont.
    \end{itemize}
\end{proof}

\begin{definition}{Quotient Map}\\
A map $q : (X,\mathcal{T}_X) \rightarrow (Y,\mathcal{T}_Y)$ is a quotient map if
\begin{itemize}
    \item $q$ is surjective;
    \item $q$ is cont;
    \item for any $U \subseteq Y$ satisfying that $q^{-1}(U)$ is open in $X$, $U$ is open in $Y$.
\end{itemize}
\end{definition}

\noindent\textbf{Remark:} It's easy to show that the canonical map is a quotient map.

\begin{proposition}
    Let $q: X \rightarrow Y$ be a quotient map, and let $g: Y \rightarrow Z$ be any map. Then
    $g \;\; is\;\; cont \Leftrightarrow g\circ q \;\; is \;\; cont.$
\end{proposition}

\begin{proof}
    Similar for proposition 4.5.
\end{proof}

\noindent\textbf{Example:} Consider $X = [0,1]$ and define $x_1\sim x_2$ if
$$(x_1,x_2) = (0,1) \;\; or \;\; (1,0).$$
Then $X= \{0,1\} \cup (\bigcup _{x\in (0,1)} \{ x\})$. Then intuitively, $X \cong S^1$ are homeomorphic.
\begin{proof}
    Define $f : [0,1] \rightarrow S^1$ by $f(t) := (\cos 2\pi t , \sin2\pi t)$. Then $f(0) = f(1) = (1,0)$. Hence, $f$ can induce a well-defined fct $g: [0,1] / \sim \;\rightarrow S^1$ with $g([t]) := f(t)$, i.e.: $f= g\circ p$. By the continuity of $f$ and proposition 4.7, we have $g$ is cont.\\
    Therefore, now we have: $g$ is cont and bijective, $S^1$ is Hausdorff, and $[0,1]$ is cpt. Then by theorem 3.30, $[0,1] / \sim \; \cong S^1$.
\end{proof}

\begin{proposition}
    Let $f: (X,\mathcal{T}_X) \rightarrow(Y,\mathcal{T}_Y)$ be a surjective cont map, and let $\sim$ be an equivalence relation on $X$ defined by the partition $\{ f^{-1} (y) \mid y \in Y\}$ (i.e.: $x\sim x'$ iff $f(x) = f(x')$). If $X$ is compact and $Y$ is Hausdorff, then $X/\sim \; \cong Y$ are homeomorphic.
\end{proposition}

\noindent\textbf{Example:}
\begin{itemize}
    \item Consider $X = [0,1] \times [0,1]$ and define $(s_1 ,t_1) \sim (s_2,t_2)$ if
    \begin{itemize}
        \item $(s_1 ,t_1) = (s_2,t_2)$;
        \item $\{ s_1,s_2\} = \{0,1\}$, $ t_1=t_2$;
        \item $\{t_1,t_2\} = \{0,1\}$, $s_1=s_2$;
        \item $\{ s_1,s_2\} = \{t_1,t_2\} = \{0,1\}$.
    \end{itemize}
    Define $f: X \rightarrow \mathbb{T}^2$ by $f(t_1,t_2) := (e^{2\pi it_1}, e^{2\pi it_2})$. Then we have: $f$ is surjective and cont, $\mathbb{T}^2$ is Hausdorff, $X$ is cpt, and 
    $$\begin{cases}
        (s_1,t_1) \sim (s_2,t_2) \Rightarrow f(s_1,t_1)=f (s_2,t_2)\\
        f(s_1,t_1)=f (s_2,t_2) \Rightarrow (s_1,t_1) \sim (s_2,t_2).
    \end{cases}$$
    Then by proposition 4.8, $X/\sim \; = [0,1]\times [0,1] /\sim \; \cong \mathbb{T}^2$.
    \item Consider $\mathbb{D}^2 := \{ (x,y) \in \mathbb{R}^2 \mid x^2 + y^2 \leq 1 \}$, and define $(x_1,y_1) \sim (x_2,y_2)$ by:
    \begin{itemize}
        \item $(x_1,y_1) = (x_2,y_2)$;
        \item $(x_1,y_1) , (x_2,y_2)\in \partial \mathbb{D}^2$.
    \end{itemize}
    Define $f:\mathbb{D}^2 \rightarrow \mathbb{S}^2$, where $\mathbb{S}^2 := \{ (x,y,z) \in \mathbb{R}^3 \mid x^2+y^2+z^2=1 \}$, by: 
    \begin{align*}
        &f(0,0) := (0,0,1)\\
        &f(x,y) := (\frac{x}{\sqrt{x^2 +y^2}} \sin (\pi\sqrt{x^2 +y^2}), \frac{y}{\sqrt{x^2 +y^2}} \sin (\pi \sqrt{x^2 +y^2}) , \cos(\pi \sqrt{x^2 +y^2})).
    \end{align*}
    Then we have: $f$ is surjective and cont, $\mathbb{S}^2$ is Hausdorff, $\mathbb{D}^2$ is cpt, and
    $$\begin{cases}
        (x_1,y_1) \sim (x_2,y_2) \Rightarrow f(x_1,y_1)=f (x_2,y_2)\\
        f(x_1,y_1)=f (x_2,y_2) \Rightarrow (x_1,y_1) \sim (x_2,y_2).
    \end{cases}$$
    Then by proposition 4.8, $\mathbb{D}^2 \cong \mathbb{S}^2$ are homeomorphic.
\end{itemize}

\begin{proposition}
    Suppose $q : X \rightarrow Y$ is a quotient map, and that $\sim$ is an equivalence relation on $X$, given by the partition $\{ q^{-1} (y) \mid y \in Y\}$. Then $X/\sim \; \cong Y$ are homeomorphic.
\end{proposition}

\begin{proof}
    Construct $h : X/\sim \;\rightarrow Y$ by $h([x]) := q(x)$. Then $h$ is bijective and well-defined. By proposition 4.7, since $q = h\circ p$ is cont, then $h$ is cont.\\
    Now we need to show that $h^{-1}$ is cont.\\
    For any open $\widetilde{U} \subseteq X/\sim$, $q^{-1}(h(\widetilde{U})) = p^{-1} h^{-1} (h (\widetilde{U})) = p^{-1} (\widetilde{U})$ is open, by the def of quotient topology. Then by the def of quotient map, $h(\widetilde{U}) = (h^{-1})^{-1} (\widetilde{U})$ is open, i.e.: $h^{-1}$ is cont.
\end{proof}

\noindent\textbf{Example:} Consider $X =\mathbb{R}$ and define $\sim$ by $x \sim y \Leftrightarrow x-y \in \mathbb{Z}$. Then consider $q : \mathbb{R} \rightarrow \mathbb{S}^1$ with $q(x) := (\cos 2\pi x, \sin 2\pi x)$. Then $q$ is a cont surjective open map. Therefore, $\mathbb{R} / \mathbb{Z := \mathbb{R}/\sim \; \cong \mathbb{S}}^1$.

\begin{definition}{Closed Equivalence Relation}\\
Let $X$ be a topological space and let $\sim$ be an equivalence relation on $X$. The relation is called \textit{closed} if
$$\sim = \{ (x,y) \in X \times X \mid x \sim y \}$$
is closed in $X \times X$.
\end{definition}

\begin{proposition}{Quotient Topology-Properties Proposition}\\
Let $\sim$ be an equivalence relation on a topological space $X$.
\begin{enumerate}
    \item If $X$ is compact, then so is $X/\sim$.
    \item If $X$ is (path-)connected, then so is $X/\sim$.
    \item If $X$ is Hausdorff and compact and if $\sim$ is closed, then $X/\sim$ is Hausdorff.
\end{enumerate}
\end{proposition}

\begin{proof}
    \textit{1,2} immediately holds by the continuity of $p$.
    \textit{3.} \begin{itemize}
        \item \textbf{Claim 1:} Given any closed subset $C$ of $X$, $p(C)$ is closed in $X/\sim$.
        \item \begin{proof}
            By the def of quotient topology, it suffices to show that $p^{-1}(p(C))$ is closed in $X$. Let $p_1,p_2: X\times X \to X$ be the projection maps onto the first respectively the second factor. Note that
            $$p^{-1}(p(C)) = \{y \in X\mid y \sim x\; for\; some\;x \in C \} = p_2 (p_1^{-1}(C) \cap \sim)$$
            Since $p_1^{-1}(C) = C \times X$ is closed in $X \times X$ ($C$ and $X$ are closed in $X$), then $p_1^{-1} (C) \cap \sim$ is closed in $X \times X$. Since $X$ is compact, then so is $X \times X$ and hence so is $p_1^{-1} (C) \cap \sim$. Therefore, $p^{-1} (p(C)) = p_2 (p_1^{-1} \cap \sim)$ is compact (by the continuity of $p_2$) and hence it is closed in $X$.
        \end{proof}
    \end{itemize}
    Now let $[x],[y] \in X/\sim$ be two disjoint points in $X/\sim$. By the Hausdorff-ness of $X$, $\{x\}$ and $\{y\}$ are closed in $X$. Hence, by claim 1,
    $$\{[x]\} = p(\{x\}) \;\;and\;\; \{[y]\} = p(\{y\})$$
    are closed in $X/\sim$. By the continuity of $p$, 
    $$p^{-1}([x]) = p^{-1} (\{[x]\}) \;\;and\;\; p^{-1}([y]) = p^{-1} (\{[y]\})$$
    are two disjoint closed subsets of $X$. Since $X$ is Hausdorff and compact, then $X$ is normal and hence $\exists$ open neighborhoods $U,V$ of $p^{-1} ([x]), p^{-1} ([y])$ respectively, s.t.: $U\cap V=\emptyset$.
    \begin{itemize}
        \item \textbf{Claim 2:} $(X/\sim) \setminus p(V^C)$ and $(X/\sim) \setminus p(U^C)$ are disjoint open neighborhoods of $[x]$ and $[y]$ in $X /\sim$.
        \item \begin{proof}
            By claim 1, $U^C,V^C$ closed can imply that $p(V^C),p(U^C)$ are closed. As for the "disjoint-ness",
            \begin{align*}
                ((X/\sim)\setminus p(U^C)) \cap ((X/\sim) \setminus p(V^C)) &= (X/\sim) \setminus (p(U^C) \cup p(V^C)) = (X/\sim) \setminus p(U^C \cap V^C) \\
                &= (X/\sim) \setminus p(X \setminus (U\cap V)) = (X/\sim) \setminus p(X) \overset{p\;surj}{=} \emptyset
            \end{align*}
            Since $p^{-1} ([x]) \cap V^C =\emptyset$, then $[x] \in (X/\sim) \setminus p(V^C)$. Similarly, $[y] \in (X/\sim) \setminus p(U^C)$. Therefore, by claim 2, we're done.
        \end{proof}
    \end{itemize}
\end{proof}

\begin{proposition}{Twice Quotient Proposition}\\
\begin{enumerate}
    \item Let $X$ be a topological space, let $\sim_1$ be an equivalence relation on $X$ and let $\sim_2$ be an equivalence relation on $X/\sim_1$. Given $x,y \in X$, we define
    $$x \sim y :\Leftrightarrow [x]_{\sim_1} \sim_2 [y]_{\sim_1}$$
    \begin{enumerate}
        \item $\sim$ gives an equivalence relation.
        \item The map $X/\sim \to (X/\sim_{1})/\sim_2$ given by $[x] \mapsto [[x]_{\sim_1}]_{\sim_2}$ is a homeomorphism.
    \end{enumerate}

    \item Let $f:X \to Y$ be a continuous map between two topological spaces and let $\sim_X$ and $\sim_Y$ be equivalence relations on $X$ and $Y$ respectively. If $x_1 \sim_X x_2$ implies $f(x_1) \sim_Y f(x_2)$, then the map
    $$\substack{X/\sim_X \;\to\; Y/\sim_Y\\
    [x] \;\mapsto\; [f(x)]}$$
    is well-defined and continuous.
\end{enumerate}
\end{proposition}

\begin{definition}{Quotients by subspace}\\
Let $X$ be a topological space and let $A \subseteq X$ be a subset. If $A= \emptyset$, then we define
$$X / \emptyset := X \sqcup \{*\}$$
Otherwise, we define $\sim_A$ by
$$x \sim_A y :\Leftrightarrow x=y\;\; or\;\; x,y \in A$$
which can easily be verified an equivalence relation on $X$. Then we define
$$X/ A := X /\sim_A$$
equipped with the quotient topology.
\end{definition}

\noindent\textbf{Remark:} \begin{itemize}
    \item $X /A \cong (X\setminus A) \cup \{[A]\}$ are homeomorphic.
    \item Given any subset $Z \subseteq X/A = (X\setminus A) \cup \{[A]\}$, we have
    $$p^{-1}(Z) =\begin{cases}
        Z &,if\;[A] \notin Z\\
        Z \cup A &,if\; [A] \in Z
    \end{cases}$$
\end{itemize}

\begin{proposition}{Cylinder Quotient-Ball Proposition}\\
\begin{itemize}
    \item The map
    $$\substack{([0,1] \times \mathbb{S}^n)/(\{0\} \times \mathbb{S}^n) \;\to\;\overline{\mathbb{B}}^{n+1}\\
    [(r,v)] \;\mapsto\; rv}$$
    is a homeomorphism.
    \item Let $X$ be a topological space and let $f: [0,1] \times \mathbb{S}^n \to X$ be a continuous map. If $f(0,v) = f(0,w)$ for any $v,w \in \mathbb{S}^n$, then the map
    $$\substack{\overline{\mathbb{B}}^{n+1} \;\to\; X\\
    r\cdot v\;\mapsto\; f(r,v)}$$
    is well-defined and continuous.
\end{itemize}
\end{proposition}

\section{Group Actions on Topological Spaces}

\begin{definition}
    Let $X$ be a set and let $G$ be a group.
    \begin{itemize}
        \item An \textit{(group) action} of $G$ on $X$ is a map
        $$\substack{G \times X \;\to\; X\\
        (g,x) \;\mapsto\; g\cdot x}$$
        with the following axioms: for any $x \in X$, any $g,h \in G$,
        $$e \cdot x =x, (g\cdot h) \cdot x= g \cdot (h\cdot x)$$

        \item The action is called \textit{free} if $g\cdot x =x$ for some $x \in X$ implies that $g=e$.

        \item If $X$ is a topological space, then we say that the action is \textit{continuous} if for any $g\in G$, the map
        $$\substack{X \;\to\; X\\
        x \;\mapsto\; g \cdot x}$$
        is continuous.

        \item Let $X$ be a topological space and let $G$ be a group that acts on $X$. We define an equivalence relation on $X$ by
        $$x \sim_G y :\Leftrightarrow g \cdot x=y \;for\;some\;g \in G$$
        Then we define
        $$X/G := X/\sim_G$$
        equipped with the quotient topology.
    \end{itemize}
\end{definition}

\begin{proposition}{Torus-as-Quotient Proposition}\\
\begin{enumerate}
    \item The map
    $$\substack{\mathbb{Z}^n \times \mathbb{R}^n \;\to\; \mathbb{R}^n\\
    (z,v) \;\mapsto\; z+v}$$
    defines a free and continuous action of the group $(\mathbb{Z}^n ,+)$ on the topological space $X= \mathbb{R}^n$.

    \item The map
    $$\varphi: \substack{\mathbb{R}^n/\mathbb{Z}^n \;\to\; (\mathbb{S}^1)^n\\
    [(t_1,\cdots,t_n)] \;\mapsto \; (\exp (2\pi it_1), \cdots, \exp (2\pi it_n))}$$
    is a homeomorphism.
\end{enumerate}
\end{proposition}

\begin{definition}{Proper Action}\\
Let $X$ be a topological space and let $G$ be a group that acts on $X$. We say that $G$ acts \textit{properly} if $\forall$ two points $x,y \in X$, $\exists$ open $U \ni x,V\ni y$, s.t.:
$$|\{g \in G \mid gU \cap V\neq \emptyset\} | < \infty$$
\end{definition}

\noindent\textbf{Example:} \begin{itemize}
    \item The action of any finite group is proper.
    \item Consider the action of $G=\mathbb{Z}$ on $X= \mathbb{R} \times [0,1]$ given by
    $$\substack{\mathbb{Z} \times (\mathbb{R} \times [0,1]) \;\to\; \mathbb{R} \times [0,1] \\
    (n,(x,y)) \;\mapsto\; (x+n,y)}$$
    which is proper. For example, given $(x,s), (y,t) \in \mathbb{R}\times [0,1]$,
    $$U:= \{(a,b) \in \mathbb{R} \times [0,1] : |a-x| <1\} \;\;and\;\; V:= \{(a,b) \in \mathbb{R} \times [0,1] : |b-y| <1\}$$
    have the desired properties in the def of proper action.
\end{itemize}

\begin{proposition}{Group Action-Hausdorff Quotient Proposition}\\
Let $G$ be a group that acts on a Hausdorff topological space $X$. If the action is proper and continuous, then $X/G$ is Hausdorff.
\end{proposition}

\begin{proof}
    \begin{lemma}{Group Action-Canonical Projection Map Lemma}\\
    Let $X$ be a topological space and let $G$ be a group that acts continuously on $X$.
    \begin{enumerate}
        \item The canonical projection map $p$ is open.
        \item If $A \subseteq X$ is a closed subset with $GA =A$, then $p(A)$ is closed in $X /G$.
    \end{enumerate}
    \end{lemma}

    \begin{proof}
        \begin{enumerate}
            \item For any open $U$ in $X$, it suffices to show that $p^{-1}(p(U))$ is open in $X$. Note that
            $$p^{-1}(p(U)) = \bigcup_{g \in G} gU$$
            Since $G$ acts contly, then the map $x \mapsto gx$ is a homeomorphism. Hence, $gU$ is open for any $g \in G$. Therefore, $p^{-1}(p(U))$ is open in $X$.

            \item Note that $$(X/G) \setminus p(A) = p(X\setminus A)$$
            Since $X \setminus A$ is open, then by 1, so is $p(X\setminus A)$.
        \end{enumerate}
    \end{proof}

    \begin{lemma}{Disjoint Translated Neighborhoods Lemma}\\
    Let $G$ be a group that acts continuously and properly on a topological space. If $X$ is Hausdorff, then $\forall$ two points $a,b\in X$, $\exists$ open $A \ni a, B\ni b$, s.t.: for any $g \in G$ with $ga=b$, we have
    $$gA \cap B =\emptyset$$
    \end{lemma}

    \begin{proof}
        Since the action is proper, then there exists open $U \ni a,V \ni b$, s.t.:
        $$|\{ g \in G \mid gU \cap V \neq \emptyset \}| < \infty$$
        Denote the above set as $\{g_1 ,\cdots, g_n\}$ and then set $I:= \{ i \in \{1,\cdots, n\} \mid g_i a=b\}$. By the Hausdorff-ness of $X$, for any $i \in I$, there exists open $U_i \ni g_ia,V_i \ni b$, s.t.: $U_i \cap V_i = \emptyset$. Now take
        $$A:= U \cap (\bigcap_{i \in I} g^{-1}_i U_i) \;\;and\;\; B:= V \cap (\bigcap_{i \in I} V_i)$$
        which are the intersection of finitely many open subsets and hence which are open. Therefore, $A$ is an open neighborhood of $a$ and $B$ is an open neighborhood of $b$. 
        
        Let $g \in G$ with $ga\neq b$. If $g \notin \{g_1 ,\cdots, g_n\}$, then $gU \cap V =\emptyset$ and hence $gA \cap B =\emptyset$. Otherwise, $g=g_i$ for some $i \in I$. Then
        $$g_i(g_i^{-1}U) \cap V_i = U_i \cap V_i =\emptyset \Rightarrow gA\cap B= g_i A \cap B=\emptyset$$
    \end{proof}

    \begin{lemma}{Disjoint Quotient Lemma}\\
    Let $G$ be a group acts on a set $X$. Let $A$ and $B$ be subsets of $X$. Then
    $$p(A) \cap p(B) =\emptyset \Leftrightarrow \forall g \in G, \;we\;have\; gA \cap B=\emptyset$$
    or equivalently
    $$p(A) \cap p(B) \neq \emptyset \Leftrightarrow \exists g \in G, s.t.: gA \cap B \neq \emptyset$$
    \end{lemma}

    \begin{proof}
        We prove the second statement:
        \begin{align*}
            p(A) \cap p(B) \neq \emptyset &\Leftrightarrow p(a) = p(b) \;for\;some\; a\in A,b\in B\\
            &\Leftrightarrow ga=b \;for\;some\;a\in A,b\in B,g \in G\\
            &\Leftrightarrow g A\cap B \neq \emptyset \;for\;some\;g \in G
        \end{align*}
    \end{proof}

    Now we back to the proof of the Group Action-Hausdorff Quotient Proposition. For any $x \neq y \in X/G$, there exist $a,b \in X$, s.t.:
    $$p(a)=x  \neq p(b) =y$$
    Hence, by the def of $\sim_G$, for any $g \in G$, $ga \neq b$. By the Disjoint Translated Neighborhoods Lemma, there exist open $A\ni a,B\ni b$, s.t.: $\forall g \in G$
    $$gA \cap B= \emptyset $$
    Then by the Disjoint Quotient Lemma, 
    \begin{equation*}
        p(A) \cap p(B) =\emptyset \tag{*}
    \end{equation*}
    By the open-ness of $p$ (Group Action-Canonical Projection Map Lemma.1) and $A,B$, then $p(A)$ and $p(B)$ are open. Therefore, $p(A)$ and $p(B)$ are respectively the disjoint open neighborhoods of $x=p(a)$ and $y=p(b)$.
\end{proof}


\noindent\textbf{Remark} for the Group Action-Canonical Projection Map Lemma: Even if $G$ acts contly on $X$, it is NOT true in general that the projection $p: X \to X/G$ is closed. For example, let $X=\mathbb{R}$ and $G= \mathbb{Z}$ with the additive action on $\mathbb{R}$. The set $A:=\{ n+\frac{1}{n} \mid n \in \mathbb{Z} \}$ is closed in $X =\mathbb{R}$ but $p(A)$ is \textbf{NOT} closed in $X/G = \mathbb{R} / \mathbb{Z}$.









\begin{proposition}{Mobius Band-as-Quotient Proposition}\\
\begin{enumerate}
    \item The map 
    $$\substack{\mathbb{Z} \times (\mathbb{R} \times [-1,1]) \;\to\; \mathbb{R} \times [-1,1] \\
    (n,(x,y)) \;\mapsto\; (x+n, (-1)^ny)}$$
    defines an action of the group $(\mathbb{Z},+)$ on the topological space $X=\mathbb{R} \times [-1,1]$ that is free, continuous and proper.

    \item Consider $([0,1] \times [0,1])/\sim$ with $(0,y) \sim (1,1-y)$, giving a mobius band. The map
    $$\substack{([0,1] \times [0,1])/\sim \;\to\; (\mathbb{R} \times [-1,1])/\mathbb{Z}\\
    [(x,y)] \;\mapsto\; [(x,2y-1)]}$$
    is a homeomorphism.

    \item The mobius band $M= ([0,1] \times [0,1])/\sim$ is compact and Hausdorff.
\end{enumerate}
\end{proposition}

\section{Projective Spaces}

\begin{definition}{Projective Spaces}\\
\begin{itemize}
    \item The map
    $$\substack{(\mathbb{R} \setminus \{0\}) \times (\mathbb{R}^{n+1} \setminus \{0\}) \;\to\; \mathbb{R}^{n+1}\setminus \{0\}\\
    (z,P) \;\mapsto\; z\cdot P}$$
    defines a continuous action of $(\mathbb{R} \setminus \{0\},\cdot)$ on $\mathbb{R}^{n+1} \setminus \{0\}$. The quotient space $\mathbb{R}^{n+1} \setminus \{0\} / \mathbb{R} \setminus \{0\}$ is called the \textit{n-dimensional real projective space $\mathbb{R}P^n$}. Particularly, we often call the \textit{real projective plane} the 2-dimensional real projective space$\mathbb{R}P^2$.

    \item We define the \textit{n-dimensional complex projective space $\mathbb{C}P^n := (\mathbb{C}^{n+1} \setminus \{0\}) / (\mathbb{C} \setminus \{0\})$} by replacing "$\mathbb{R}$" by "$\mathbb{C}$".
\end{itemize}
\end{definition}

\noindent\textbf{Remark} (\textit{Notation}): For $(x_0,\cdots,x_n) \in \mathbb{R} \setminus \{0\}$, we denote by $[x_0 :\cdots: x_n]$ the corresponding point in $\mathbb{R}P^n$. Similarly for points in $\mathbb{C}P^n$.

\begin{proposition}{Projective Space-Sphere Quotient Proposition}\\
Let $n \in \mathbb{N} \cup \{0\}$.
\begin{enumerate}
    \item The map
    $$\substack{\{ \pm 1\} \times \mathbb{S}^n \;\to\; \mathbb{S}^n\\
    (w,(x_1, \cdots, x_{n+1})) \;\mapsto\; (w \cdot x_1 ,\cdots, w \cdots x_{n+1})}$$
    defines a continuous action of $\{ \pm1\}$ on $\mathbb{S}^n$. The maps
    $$\substack{\mathbb{S}^n/ \{ \pm1\} \;\to\; \mathbb{R}P^n\\
    [x] \;\mapsto\; [x]} \;\;\;and\;\;\; \substack{\mathbb{R}P^n \;\to\; \mathbb{S}^n/ \{ \pm1\}\\
    [x] \;\mapsto\; [\frac{x}{||x||}]}$$
    are inverses of one another and they are homeomorphisms.

    \item The map 
    $$\substack{\mathbb{S}^1 \times \mathbb{S}^{2n+1} \;\to\; \mathbb{S}^{2n+1}\\
    (w,(z_1 \in \mathbb{C} ,\cdots, z_{n+1} \in \mathbb{C})) \;\mapsto\; (w \cdot z_1 ,\cdots, w \cdot z_{n+1})}$$
    defines a continuous action of $(\mathbb{S}^1,\cdot)$ on $\mathbb{S}^{2n+1}$. The maps
    $$\substack{\mathbb{S}^{2n+1} /\mathbb{S}^1 \;\to\; \mathbb{C}P^n\\
    [z] \;\mapsto\; [z]} \;\;\;and\;\;\; \substack{\mathbb{C}P^n \;\to\; \mathbb{S}^{2n+1}/\mathbb{S}^1\\
    [z_1: \cdots: z_{n+1}] \;\mapsto\; [\frac{(z_1, \cdots, z_{n+1})}{\sqrt{|z_1|^2 + \cdots+ |z_{n+1}|^2}}]}$$
    are inverses of one another and they are homeomorphisms.
\end{enumerate}
\end{proposition}

\begin{proposition}{Projective Spaces Properties Proposition}\\
$\mathbb{R}P^n$ and $\mathbb{C}P^n$ are compact, path-connected and Hausdorff.
\end{proposition}

\begin{proof}
    By the previous proposition, it suffices to show that $\mathbb{S}^n /\{ \pm1\}$ and $\mathbb{S}^{2n+1}/\mathbb{S}^1$ are compact, path-connected and Hausdorff. I follows immediately from the fact that $\mathbb{S}^n$ and $\mathbb{S}^{2n+1}$ are compact and path-connected together with the Quotient Topology-Properties 1,2 that both quotients are compact and path-connected.
    
    As for the Hausdorff-ness, note that the action of $\{\pm1\}$ on $\mathbb{S}^n$ is proper and then by the Group Action-Hausdorff Quotient Prop, $\mathbb{S}^n/ \{\pm1\}$ is Hausdorff.

    Now it remains to show that $\mathbb{S}^{2n+1}/ \mathbb{S}^1$ is Hausdorff. It suffices to show that $\sim_{\mathbb{S}^1}$ is closed by the Quotient Topology-Properties Prop.3, i.e.:
    $$G:= \{(v_1,v_2) \in \mathbb{S}^{2n+1} \times \mathbb{S}^{2n+1} \mid w \cdot v_1= v_2\; for\;some\; w\in \mathbb{S}^1\}$$
    is closed in $\mathbb{S}^{2n+1} \times \mathbb{S}^{2n+1}$. Consider the map $\Phi: \mathbb{S}^1 \times \mathbb{S}^{2n+1} \to \mathbb{S}^{2n+1} \times \mathbb{S}^{2n+1}$ given by
    $$\Phi (z,v) := (v, z\cdot v)$$
    Note that $G= \Phi(\mathbb{S}^1 \times \mathbb{S}^{2n+1})$ and that $\Phi$ is continuous and $\mathbb{S}^1 \times \mathbb{S}^{2n+1}$ is compact. Therefore, $G$ is compact and hence it is closed.
\end{proof}

\begin{proposition}{Projective Space-Avatar Proposition}\\
\begin{enumerate}
    \item The maps
    $$\substack{\mathbb{S}^2 \cong \mathbb{C} \cup \{ \infty\} \;\to\; \mathbb{C}P^1\\
    z \;\mapsto\; \begin{cases}
        [z:1] &,if\;z \in \mathbb{C}\\
        [1:0] &,if \;z= \infty
    \end{cases} } \;\;\;and\;\;\; \substack{\mathbb{C}P^1 \;\to\; \mathbb{C}\cup \{ \infty\} \cong \mathbb{S}^2 \\
    [z_0:z_1] \;\mapsto\; \begin{cases}
        z_0 \cdot z_1^{-1} &,if\;z_1 \neq 0\\
        0 &,otherwise
    \end{cases} }$$
    are inverses of one another and both are homeomorphisms. Similarly we can see that
    $$\mathbb{R}P^1 \cong \mathbb{R} \cup \{\infty\} \cong \mathbb{S}^1$$
    are homeomorphic.

    \item Let $n \in \mathbb{N} \cup \{0\}$. If we denote by $\sim$ the equivalence relation on $\overline{B}^n$ that is generated by $P \sim -P$ for $P \in \mathbb{S}^{n-1} = \partial \overline{B}^n$, then the maps
    $$\substack{\overline{B}^n/ \sim \;\to \mathbb{R}P^n\\
    [x] \;\mapsto\; \; [(x, \sqrt{1- ||x||^2})] } \;\;\;and\;\;\; \substack{\mathbb{R}P^n \;\to\; \overline{B}^n/ \sim\\
    [(v,x_{n+1})] \;\mapsto\; \begin{cases}
        [v] &,if\; x_{n+1} \geq 0\\
        [-v] &,if\; x_{n+1} <0
    \end{cases} }$$
    are homeomorphisms that are inverses of one another.

    \item As in 2, let $\sim$ be the equivalence relation on $\overline{B}^3$ that is generated by $x\sim -x$ for $x \in \mathbb{S}^2$. The map
    $$F:= \substack{\overline{B}^3/\sim \;\to\; SO(3)\\
    [r\cdot v] \;\mapsto\; A_{r,v} }$$
    where $A_{r,v}$ denotes the matrix in $SO(3)$ which represents rotation by the angle $\pi \cdot r$ around the axis $\mathbb{R}_{\geq 0} \cdot v$ where the rotation direction is given by the right-hand rule, is a homeomorphism. In particular, $SO(3) \cong \mathbb{R}P^3$ are homeomorphic.
\end{enumerate}
\end{proposition}

\noindent\textbf{Remark:} Now we get the identifications:
$$\mathbb{S}^1 =_i \mathbb{R}P^1, \mathbb{S}^2 =_i \mathbb{C}P^1, \mathbb{R}P^n=_i \overline{B}^n/\sim$$
with $\sim$ defined in the second statement of the above proposition (Projective Space-Avatar Proposition).

\section{Simplicial Complex}

\begin{definition}{n-simplex}\\
$$ \Delta ^n :=\{ (x_1, \cdots , x_{n+1}) \in \mathbb{R}^{n+1} \mid x_i \geq 0, \sum _{i=1}^{n+1} x_i =1 \}$$
\begin{itemize}
    \item Vertices of $\Delta ^n := \{ (x_1, \cdots , x_{n+1}) \in  \Delta^n \mid x_i = 1$ for some $i \}$.
    \item For any given $\mathcal{A} \subseteq \{ 1, \cdots ,n+1 \}$, the face of $\Delta ^n$ is defined by $\{ (x_1, \cdots , x_{n+1}) \in \Delta^n \mid x_i = 0, \forall i \in \mathcal{A} \}$.
    \item Interior of $\Delta ^n := \{ (x_1, \cdots , x_{n+1}) \in  \Delta^n \mid x_i \neq 0, \forall i \}$.
\end{itemize}
\end{definition}

\begin{definition}{Face Inclusion}\\
Let $m<n$. A face inclusion of $\Delta^m$ into $\Delta ^n$ is an injective map $f: \Delta^m \rightarrow \Delta^n$ which maps vertices to vertices and is the restriction of an injective linear map $T : \mathbb{R}^{m+1} \rightarrow \mathbb{R} ^{n+1}$.
\end{definition}

\noindent\textbf{Remark:}\\
$\{$all face inclusion$\}$ $\xleftrightarrow{1:1}$ $\{$injective map $g: \{ 1,\cdots, m+1 \} \rightarrow  \{ 1,\cdots, n+1 \}$$\}$.

\noindent\textbf{Example:} Linear map $f: \mathbb{R}^2 \rightarrow \mathbb{R} ^3$ with $T(1,0) = (0,1,0)$ and $T(0,1) = (0,0,1)$ is a face inclusion.

\begin{definition}{Simplicial Complex}\\
A simplicial complex is a pair $K= (V, \Sigma)$, where
\begin{itemize}
    \item $V$ is a (finite most of times) set of vertices;
    \item $\Sigma$ is a collection of non-empty finite subsets of $V$, called simplices, satisfying:
    \begin{itemize}
        \item For any $v \in V$, $\{ v \} \in \Sigma$;
        \item If $\sigma \in \Sigma$, then $(P(\sigma) - \{ \emptyset \}) \subseteq \Sigma$.
    \end{itemize}
\end{itemize}
\end{definition}

\noindent\textbf{Example:} If $V = \{ 1,2,3,4\}$, then one can take:
$$\Sigma = \{ \{ 1\}, \cdots, \{ 4\}, \{ 1,3,4\}, \{ 1,3\} , \{ 3,4\}, \{ 1,4\} \}.$$

\begin{definition}{Topological Realization}\\
Let $K = (V , \Sigma)$ be a simplicial complex. The topological realization of $K$ is a topological space, denoted by $|K|$ or $|(V,\Sigma)|$, given by identifying $\Delta ^n$ simplices, i.e.:
\begin{itemize}
    \item For any $\sigma \in \Sigma$ with $|\sigma| = n+1$, take a copy of n-simplex and denote it as $\Delta_{\sigma}$;
    \item Whenever $\sigma \subseteq \tau \in \Sigma$, identify $\Delta_\sigma$ with a face of $\Delta_\tau$ through face inclusion.
\end{itemize}
Or equivalently, $|K|$ is a quotient space of the disjoint union $\bigsqcup _{\sigma \in \Sigma} \Delta_\sigma$, by the equivalence relation which identifies a point $y\in \sigma$ with its image under the face inclusion $\sigma \rightarrow \tau$, for any $\sigma \subseteq \tau$.
\end{definition}

\begin{definition}{Triangulation}\\
A triangulation of a topological space $X$ is a simplicial complex $K = (V,\Sigma)$ together with a choice homeomorphism $f: |K| \xrightarrow{\cong} X$.
\end{definition}

\begin{definition}{Euler Characteristic}\\
For any simplicial complex $(V,\Sigma)$, the Euler characteristic is defined by
$$\chi (V,\Sigma) := \sum_{i=1}^\infty (-1)^i \cdot \#\{ subsets\;\;in\;\;\Sigma\;\;with\;\;(i+1)-element \}.$$
\end{definition}

\begin{proposition}
    $$\chi(V_1,\Sigma_1) = \chi(V_2,\Sigma_2) \Leftrightarrow |(V_1,\Sigma_1)| = |(V_2,\Sigma_2)|$$
\end{proposition}

\begin{definition}
    For any topological space $X$ with $X \cong |(V,\Sigma)|$, by proposition 4.16, we can define
    $$\chi (X) := \chi (V,\Sigma).$$
\end{definition}

\section{Properties of Simplicial Complex}

\begin{definition}{Simplicial Subcomplex}\\
A subcomplex of a simplicial complex $K = (V,\Sigma)$ is a simplicial complex $K' = (V',\Sigma')$ such that
$$V' \subseteq V, \Sigma' \subseteq \Sigma.$$
\end{definition}

\begin{proposition}
    Let $K' = (V',\Sigma')$ be a subcomplex of $K=(V,\Sigma)$ and let $i: (X':=) \bigsqcup _{\sigma' \in \Sigma'} \sigma' \rightarrow (X:=) \bigsqcup_{\sigma\in \Sigma}\sigma$ be an inclusion map. Then we can get a map
    $$\widetilde{i} := (p')^{-1} \circ i \circ p \;\; with\;\; \widetilde{i}([x']) = [i(x')].$$
    Then the map $\widetilde{i}$ is injective, well-defined, and continuous.\\
    Moreover, $\widetilde{i} (|K'|) $ is closed in $|K|$.
\end{proposition}

\begin{proof}
    Note that for $x',y' \in X'$, 
    $$x' \sim y' \Leftrightarrow i(x') \sim i(y').$$
    Hence, $\widetilde{i}$ is injective and well-defined. Also, since $p\circ i: X' \rightarrow X/\sim$ is cont, then $\widetilde{i}$ is cont.

    To see that $\widetilde{i} (|K'|) $ is closed in $|K|$, by the def of $\widetilde{i}$, we only need to show that $p\circ i(X')$ is closed in $|K|$:\\
    For any $y \in p^{-1}((p\circ i)(X'))$, $y \in \Delta_\tau$ for some $\tau \in \Sigma$. Then $y \in \Delta_\tau$ must lie on a face of $\Delta_\tau$ which is "glued" to $\Delta_{\tau \cap \sigma'}$ by face inclusion for some $\sigma' \in \Sigma '$.\\
    Let $\Delta_{\tau \cap \Sigma'}$ be the union of faces of $\Delta_\tau$ given by face inclusion $\Delta_{\tau \cap \sigma'} \overset{i}{\rightarrow} \Delta_{\tau}$ for some $\sigma' \in \Sigma'$. Then $\Delta_{\tau \cap \Sigma'}$ is closed in $\Delta_\tau$ (a face on $\Delta_\tau$ is the zero set of some linear equations $f_1, \cdots,f_n$, so it is equal to $\bigcap_{i=1}^n f_i^{-1}(0)$,) and $p^{-1} (p \circ i (X')) = \bigsqcup_{\tau \in \Sigma} \Delta_{\tau \cap \Sigma'}$ is closed. Then by the def of $p$, canonical projection map, we're done
\end{proof}

\begin{definition}
    Let $K = (V,\Sigma)$ be a simplicial complex and let $V' \subseteq V$. Then the subcomplex spanned by $V'$ is $(V',\Sigma')$ such that
    \begin{itemize}
        \item $V'$ denotes the vertex set;
        \item the simplices $\Sigma'$ is given by $$\{ \sigma \in \Sigma \mid \sigma \subseteq V' \}.$$
    \end{itemize}
\end{definition}

\begin{definition}
    Let $K= (V,\Sigma)$ be a simplicial complex.
    \begin{itemize}
        \item (\textbf{Link}): The link of $v \in V$, denoted as $lk(v)$, is the subcomplex with vertex set
        $$\{ w\in V-\{v\} \mid \{v,w\} \in \Sigma \}$$
        and simplices
        $$\{ \sigma \in \Sigma \mid v \notin \sigma, \sigma \cup \{v\} \in \Sigma \}.$$
        \item (\textbf{Star}): The star of $v$, denoted as $st(v)$, is
        $$\bigcup_{\sigma \in \Sigma,v \in \sigma} inside(\sigma),$$
        where $inside(\Delta^n) := \{ (x_1,\cdots,x_{n+1}) \in \Delta^n \mid x_i > 0, \forall i \}$ for n-simplex $\Delta^n$.
    \end{itemize}
\end{definition}

\begin{proposition}
    $st(v)$ is open and $v \in st(v)$.
\end{proposition}

\begin{proposition}
    Let $K =(V,\Sigma)$ with $|V| < \infty$. Then $K$ is compact.
\end{proposition}

\begin{proof}
    Let $p: D\rightarrow K$ be a canonical map, which is cont. Since $D$, the finite disjoint union of $\Delta_\sigma$'s, is compact, then $p(D) = |K|$ is compact.
\end{proof}

\begin{proposition}
    For any $K=(V,\Sigma)$ with $|V|<\infty$, there is a continuous injection: for some $n$,
    $$f: |K| \rightarrow \mathbb{R}^n.$$
\end{proposition}

\begin{proof}
    Let $K^\Pi := (V,\Sigma^\Pi)$ where $\Sigma^\Pi = P(V)$. Then
    $$|K^\Pi| = \Delta^{|V|-1} \subseteq \mathbb{R}^{|V|}.$$
    Consider $D := \bigsqcup_{\sigma\in \Sigma} \Delta_\sigma$ and $D':= \bigsqcup_{\Delta_{\Sigma^\Pi}\in \Sigma^\Pi} \Delta_{\Sigma^\Pi}$ in $(V,\Sigma)$ and $(V,\Sigma^\Pi)$. Then construct the map $\widetilde{i} : D \overset{i}{\rightarrow} D' \overset{p'}{\rightarrow} |K|$, which can descend to an inclusion $i :D/\sim \; \rightarrow |K^\Pi|$.
    Therefore, $|K| \overset{i}{\rightarrow} |K^\Pi|$.
\end{proof}

\begin{corollary}
    If $K=(V,\Sigma)$ with $|V|<\infty$, then $|K|$ is Hausdorff.
\end{corollary}
\begin{proof}
    Let $g: |K| \rightarrow \mathbb{R}^n$. Consider the bijective $g: |K| \rightarrow g(|K|)$, which is cont. Since $|K|$ is compact, and $g(|K|)$ is Hausdorff, then $|K| \cong g(|K|)$. Therefore, $|K|$ is Hausdorff.
\end{proof}

\begin{definition}{Edge Path}\\
An edge path of $K=(V,\Sigma)$ is a sequence of vertices $(v_1,\cdots,v_n)$, $v_i \in V$, s.t.: $\{ v_i ,v_{i+1} \} \in \Sigma$ for all $i$.
\end{definition}

\begin{proposition}
    Let $K=(V,\Sigma)$ be a simplicial complex. The followings are equivalent:
    \begin{enumerate}
        \item $|K|$ is connected;
        \item $|K|$ is path-connected;
        \item Any two vertices $V$ can be joined by an edge path, i.e.: $\forall u,v \in V$, $\exists v_1,\cdots,v_k \in V$, s.t.: $(u,v_1,\cdots,v_k,v)$ is an edge path.
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{itemize}
        \item (2 $\Rightarrow$ 1): Clearly.
        \item (3 $\Rightarrow$ 2): For any $y,x \in |K| = |(V,\Sigma)|$, there exist $\sigma ,\tau \in \Sigma$, s.t.: $$x \in \Delta_\sigma,y \in \Delta_\tau.$$
        Since $\Delta_\sigma,\Delta_\tau$ path-connected can join $x$ to a vertex $v_x$ in $\Delta_\sigma$, and $y$ to a vertex $v_y$ in $\Delta_\tau$. By 3, there exists a edge path joining $v_x$ to $v_y$. Hence, there exists a path joining $x$ to $y$.
        \item (1 $\Rightarrow$ 3): Suppose on the contrary that fix $v \in V$, and there exists $w \in V$, s.t.: there are NO edge path joining $v,w$. Define:\\
        $V' := \{ v' \in V \mid v,v'$ can be joined by an edge path $\}$ and \\
        $V'' := \{ v'' \in V \mid v,v''$ can NOT be joined by any edge path $\}$.\\
        Then we have $V',V'' \neq \emptyset$ (by assumption) is disjoint (by the def of $V',V''$) and $V = V' \cup V''$.\\
        Consider $K' :=$ subcomplex of $K$ spanned by $V'$ and\\
        $K'' :=$ subcomplex of $K$ spanned by $V''$.\\
        Then it's clear that $|K'|$ and $|K''|$ are disjoint in $|K|$.
        \begin{itemize}
            \item \textbf{Claim:} $|K| = |K'| \cup |K''|$.\\
            \textit{Proof of claim:} Suppose on the contrary that there exists $x \in |K| - (|K'| \cup |K''|)$. Then there exists $\pi \in \Sigma$, s.t.: $x \in \Delta_\pi$. Then $\pi$ must contain vertices in both $V'$ and $V''$, otherwise $\Delta_\pi$ is in $|K'|$ or $|K''|$. Hence, there exist $w' \in V'$, $w'' \in V''$, s.t.:
            $$w' , w'' \in \Delta_\sigma.$$
            But then $\{ w',w''\} \in \Sigma$, which means that there exists an edge path joining $w' \in V'$ and $w'' \in V''$, contradicting to the def of $V'$ and $V''$. $\square$
        \end{itemize}
        Since $|K'|$ and $|K''|$ are closed in $|K|$, then by claim, $|K''| = |K| - |K'|$ and $|K'| = |K| - |K''|$ are open in $|K|$. Therefore, $|K|$ is disconnected, contradicting 1.
    \end{itemize}
\end{proof}
