\chapter{Properties of Topological Spaces}

\section{Hausdorff Property}

\begin{definition}{First Separation Axiom, $T_1$}\\
A topological space $(X,\mathcal{T}_X)$ is said to satisfy the first separation axiom (or to be a $T_1$ space) if for every pair of distinct points $x,y \in X$, there exists an open set $U \subseteq X$, s.t.: $x \in U$ and $y \notin U$.
\end{definition}

\begin{proposition}
A topological space $X$ satisfies the first separation axiom ($T_1$) iff every singleton set $\{x\}$ is closed in $X$. 
\end{proposition}

\begin{proof}
    \begin{itemize}
        \item ($\Rightarrow$): Fix $x\in X$. For any $y \in X-\{x\}$, there is an open $U_y \ni y$ such that $x\notin U_y$. Then:
        $$X- \{x\} = \bigcup_{y\neq x}\{y\} \subseteq \bigcup _{y \neq x} U_y \subseteq X- \{x\}.$$
        Hence, $X- \{x\} = \bigcup _{y \neq x} U_y$, where $U_y$ is open. Therefore, $\{x\}$ is closed.
        \item ($\Leftarrow$): For any $x,y \in X$ with $x\neq y$, set $U:= X-\{y\}$, which is open, containing $x$, and not containing $y$. Hence, $X$ is a $T_1$ space.
    \end{itemize}
\end{proof}

\begin{definition}{Hausdorff Space}\\
A topological space $X$ is called Hausdorff (or said to satisfy the second separation axiom, $T_2$) if for every pair of distinct points $x,y \in X$, there exists open neighborhoods $U$ of $x$ and $V$ of $y$ such that $U\cap V= \emptyset$.
\end{definition}

\noindent\textbf{Example} (A topological space that is a $T_1$ space may not necessarily be second separable):\\
Consider $\mathcal{T}_{co-finite}$. Then $X$ is a $T_1$ space but NOT Hausdorff if $X$ is infinite:\\
Suppose on the contrary that for given $x \neq y$, there are two open sets $U,V$ such that $x\in U$ and $y\in V$, and let $U\cap V=\emptyset$, which implies $X = X- (U\cap V) = (X-U) \cup (X-V)$. Then the union of two finite sets equals $X$ which is infinite. Contradiction. \hfill$\square$

\begin{proposition}
    If the topological space $(X,\mathcal{T})$ is Hausdorff, then all sequences $\{x_n\}$ in $X$ has at most one limit.
\end{proposition}

\begin{proof}
    Suppose on the contrary that $\{x_n\} \rightarrow a$ and $\{x_n\} \rightarrow b$ with $a \neq b$. Then by Hausdorff-ness, there are two open $U,V \in \mathcal{T}$ such that $U\cap V =\emptyset$ and $a \in U, b\in V$.
    Then $\exists N,M>0$, s.t.: $\{x_N,x_{N+1},\cdots\} \subseteq U$ and $\{x_M,x_{M+1},\cdots\} \subseteq V$. Take $K :=\max \{M,N\}+1$. Then $x_K \in U \cap V = \emptyset$, contradiction.
\end{proof}

\begin{proposition}
    Let $X,Y$ be two Hausdorff spaces. Then the product space $X\times Y$, equipped with the product topology, is Hausdorff.
\end{proposition}

\begin{proof}
    For any $(x_1,y_1) \neq (x_2,y_2) \in X\times Y$, either $x_1\neq x_2$ or $y_1 \neq y_2$. WLOG, let $x_1 \neq x_2$. By the Hausdorff-ness, there are two sets $U,V \subseteq X$, s.t.: $x_1 \in U$, $x_2 \in V$ and $U\cap V=\emptyset$. Then $(x_1,y_1) \in U\times Y$, $(x_2.y_2) \in V\times Y$, and $(U\times Y) \cap (V\times Y)=\emptyset$.
\end{proof}

\begin{definition}{Topological Embedding}\\
A continuous injective map $f: (X,\mathcal{T}_X) \rightarrow (Y,\mathcal{T}_Y)$ is called a topological embedding if $f: X \rightarrow f(X)$ is a homeomorphism onto its image, where $f(X)$ is equipped with the subspace topology from $Y$.
\end{definition}

\noindent\textbf{Remark:} Equivalently, $f$ is an embedding if it is cont, injective, and for every open set $U \subseteq X$, there exists an open set $V \subseteq Y$ such that $U = f^{-1}(V) \cap X$.

\begin{proposition}
    Let $f: (X,\mathcal{T}_X) \rightarrow (Y,\mathcal{T}_Y)$ be a topological embedding. If $Y$ is Hausdorff, then $X$ is Hausdorff.
\end{proposition}

\begin{proof}
    $f$ is embedding implies that $f:X \rightarrow f(X)$ is homeomorphism onto its image, where $f(X)$ is equipped with the subspace topology from $Y$.
    Take distinct points $a,b \in X$. Then $f(a) \neq f(b)$ in $Y$. 
    
    By the Hausdorff-ness of $Y$, there are two open sets $U,V \subseteq Y$, s.t.: $f(a) \in U$, $f(b) \in V$, and $U \cap V=\emptyset$.
    
    Hence, $U \cap f(X)$ and $V \cap f(X)$ are the open neighborhoods of $f(a)$ and $f(b)$ in $f(X)$, respectively, with $(U\cap f(X)) \cap (V \cap f(X))= \emptyset$.
    
    Since $f: X \rightarrow f(X)$ is a homeomorphism, then $f^{-1}(U \cap f(X))$ and $f^{-1}(U \cap f(X))$ are two open neighborhoods of $a$ and $b$, respectively, with $f^{-1}(U \cap f(X))\cap f^{-1}(U \cap f(X)) = \emptyset$.
\end{proof}

\noindent\textbf{Remark:} If $f$ is only cont and injective, the conclusion may fail. For example, let $X$ be any non-Hausdorff space and let $f: X\rightarrow\{*\}$ be the constant injective map into a one-point Hausdorff space. Then $f$ is cont and injective, but $X$ is NOT Hausdorff.

\begin{corollary}
    Let $f: X \rightarrow Y$ be a homeomorphism. Then $X$ is Hausdorff iff $Y$ is Hausdorff. In particular, Hausdorff-ness is a topological property, i.e: a property preserved under homeomorphism.
\end{corollary}

\section{Connectivity}

\begin{definition}{Connected Space}\\
A topological space $(X, \mathcal{T})$ is said to be disconnected if there exist non-empty open sets $U,V \in \mathcal{T}$ such that $U \cap V = \emptyset , U \cup V = X$.\\
If no such pair exists, then $X$ is called connected.
\end{definition}

\begin{proposition}
    For a topological space $(X, \mathcal{T})$, the following are equivalent:
    \begin{enumerate}
        \item $X$ is connected.
        \item The only subsets of $X$ which are both open and closed are $\emptyset$ and $X$.
        \item Every cont fct $f: X \rightarrow \{0,1\}$ (where $\{0,1\}$ is equipped with the discrete topology) is constant.
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{itemize}
        \item (1 $\Rightarrow$ 2): Suppose $U \subseteq X$ is both open and closed. Then $U$ and $X-U$ are disjoint open sets with $U \cup (X-U) = X$. By connectivity of $X$, either $U=\emptyset$ or $X-U= \emptyset$. Hence, $U = \emptyset$ or $U=X$.
        \item (2 $\Rightarrow$ 3): Let $f: X \rightarrow \{0,1\}$ be cont. Define $U:= f^{-1}(\{0\})$ and $V:= f^{-1}(\{1\})$. Then $U,V$ are open disjoint, and $U \cap V = X$. By 2, either $(U,V)= (X,\emptyset)$ or $(U,V)= (\emptyset,X)$. Thus, $f$ is constant.
        \item (3 $\Rightarrow$ 2): Suppose $U \subseteq X$ is both open and closed. Define $$ f(x):=
        \begin{cases}
            0,& if \;\; x \in U\\
            1,&otherwise.
        \end{cases}$$
        This map $f$ is cont. Then by 3, $f$ is constant. Hence, $U =\emptyset$ or $U=X$.
        \item (2 $\Rightarrow$ 1): Suppose on the contrary that $X$ is disconnected, i.e.: there are disjoint non-empty open sets $U,V \subseteq X$ with $U \cup V=X$. Then $U=X-V$ is both open and closed, contradicting 2.
    \end{itemize}
\end{proof}

\begin{corollary}
    The interval $[a,b] \subseteq \mathbb{R}$ is connected.
\end{corollary}

\begin{proof}
    Suppose on the contrary that there exists a cont fct $f : [a,b] \rightarrow \{0,1\}$ that takes 2 values. Construct the mapping $\tilde{f} : [a,b] \overset{f}{\rightarrow} \{0,1\} \overset{i}{\rightarrow} \mathbb{R}$ with $\tilde{f} = i \circ f$.

    Note that $\{0,1\} \subseteq \mathbb{R}$ denotes the subspace topology, we imply the inclusion mapping $i : \{0,1\} \rightarrow\mathbb{R}$ with $s \mapsto s$ is cont. Then the composition of cont mappings is cont, i.e.: $\tilde{f}$ is cont. Since $f$ can take two values, there are $p,q \in [a,b]$, s.t.: $\tilde{f}(p) = (i\circ f)(p) = 0$ and $\tilde{f}(q) = (i\circ f)(q) = 1$. Then by IVT, there is $r \in [a,b]$ such that $\tilde{f} (r) = (i \circ f)(r) = 1/2$. Hence, $f(r) = 1/2$, contradiction.
\end{proof}

\begin{definition}{Connected Subset}\\
A non-empty subset $S \subseteq X$ is connected if $(S, \mathcal{T}_S)$, where $\mathcal{T} _S$ denotes the subspace topology on $S$, is a connected topological space.\\
Equivalently, $S \subseteq X$ is connected if, whenever $U,V \subseteq X$ are open sets such that $S \subseteq U\cup V$, $(U \cap V) \cap S = \emptyset$, then it follows that either $U \cap S=\emptyset$ or $V \cap S=\emptyset$.
\end{definition}

\begin{proposition}
    If $f:X \rightarrow Y$ is cont and $A \subseteq X$ is connected, then $f(A)$ is connected. In other words, the cont image of a connected set is connected.
\end{proposition}

\begin{proof}
    Suppose $U,V \subseteq Y$ are open s.t.: $f(A) \subseteq U\cup V$ and $(U \cap V) \cap f(A) = \emptyset$. Then $A \subseteq f^{-1}(U) \cup f^{-1}(V)$ and $(f^{-1}(U) \cap A) \cap (f^{-1}(V) \cap A) = \emptyset$. By the connectivity of $A$, either $f^{-1}(U) \cap A = \emptyset$ or $f^{-1}(V) \cap A = \emptyset$. Therefore, either $f(A) \cap U = \emptyset$ or $f(A) \cap V = \emptyset$.
\end{proof}

\begin{proposition}
    Let $B, A_i$ $i\in I$ be connected subsets of $X$, s.t.: $B \cap A_i \neq \emptyset$ for all $i\in I$. Then $B \cup (\bigcup_{i\in I}A_i)$ is connected.
\end{proposition}

\begin{proof}
    Consider a cont fct $f : B \cup (\bigcup_{i\in I}A_i) \rightarrow \{0,1\}$. Let $f_i := \left. f \right| _{A_i} : A_i \overset{i}{\rightarrow} B \cup (\bigcup_{i\in I}A_i) \overset{f}{\rightarrow} \{0,1\}$ and $f_B := \left. f \right| _{B} : B \overset{i}{\rightarrow} B \cup (\bigcup_{i\in I}A_i) \overset{f}{\rightarrow} \{0,1\}$. Then by the connectivity of $B, A_i$, $f_i, f_B$ are constant fcts.
    
    Suppose $f_B \equiv 1$. Then $\left. f \right|_{B \cap A_i} \equiv 1$ for all $i \in I$. Hence, $f_i \equiv 1$ for all $i\in I$. Therefore, $f \equiv 1$ is a constant fct.
\end{proof}

\begin{corollary}
    If $\{A_i\}_{i\in I}$ is a family of connected subsets of $X$ s.t.: for all $i,j \in I$,
    $$A_i \cap A_j \neq \emptyset,$$
    then the union $\bigcup_{i \in I}A_i$ is connected.
\end{corollary}

\begin{proposition}
    If $X$ and $Y$ are connected spaces, then the product space $X \times Y$, equipped with the product topology, is connected.
\end{proposition}

\begin{proof}
    Fix $y_0 \in Y$. The subset $B:= X\times \{y_0\} \subseteq X\times Y$ is homeomorphic to $X$, hence connected.For any $x \in X$, the subset $C_x := \{x\} \times Y \subseteq X\times Y$ is homeomorphic to $Y$, hence connected. Moreover, for any $x\in X$, $B \cap C_x =\{(x,y_0)\} \neq \emptyset$. Therefore, by proposition 3.14, the union $B \cup (\bigcup_{x \in X} C_x) = X\times Y$ is connected.
\end{proof}

\begin{definition}{Path-Connected}\\
Let $(X,\mathcal{T})$ be a topological space.
\begin{itemize}
    \item (\textbf{Path}): A path between two points $x,y \in X$ is a cont map $p: [0,1] \rightarrow X$ with $p(0) =x$ and $p(1)=y$.
    \item (\textbf{Path-connected Space}): $X$ is said to be path-connected if any two points in $X$ can be joined by a path.
    \item (\textbf{Path-connected Subset}): A subset $A \subseteq X$ is path-connected if it is path-connected with respect to the subspace topology. Equivalently, $A$ is path-connected for every pair of points $x,y \in A$, there is a cont map $p: [0,1] \rightarrow X$ with $p(0) =x$ and $p(1)=y$.
\end{itemize}
\end{definition}

\begin{proposition}
    All path-connected spaces are connected.
\end{proposition}

\begin{proof}
    Fix $x_0 \in X$. For any $y \in X$, by path-connected-ness, there is a cont map
    $$p_y : [0,1] \rightarrow X \;\; with \;\; p_y (0)=x_0 \;\; and\;\; p_y(1) = y.$$
    Define $C_y := p_y([0,1]) \subseteq X$. By proposition 3.13, each $C_y$ is connected. Moreover, for any $y_1,y_2\in X$, $\{x_0\} \subseteq C_{y_1} \cap C_{y_2} \neq \emptyset$. Hence, by corollary 3.15, $X = \bigcup_{y\in X} C_y$ is connected.
\end{proof}

\begin{proposition}
    If $A \subseteq B \subseteq \bar{A}$ and $A$ is connected, then $B$ is connected.
\end{proposition}

\begin{proof}
    Suppose on the contrary that $X$ is not connected. Then there are two disjoint non-empty open sets $U,V \subseteq X$, s.t.: $U \cup V =X$ (therefore, (*): $U$ and $V$ are closed in $X$).

    Then $A =A\cap X = A \cap (U \cup V) = (A \cap U) \cup (A \cap V)$. Hence, by the connectivity of $A$, WLOG, $A \subseteq V$. Then $X =\bar{A} \cap X \subseteq \bar{V} \cap X \overset{(*)}{=} V$, contradiction.
\end{proof}

\noindent\textbf{Example} (Topologist's comb):\\
The converse of proposition 3.18 is false.\\
Consider the subset $X \subseteq \mathbb{R}^2$ defined by 
$$X = ([0,1] \times \{0\}) \cup (\bigcup_{n \in \mathbb{N}_+} (\{\frac{1}{n}\} \times [0,1])) \cup \{(0,1)\}$$
called Topologist's comb, which is connected but NOT path-connected.
\begin{itemize}
    \item $X$ is NOT path-connected:\\
    Suppose that there is a path $p: [0,1] \rightarrow X$ with $p(0) = (0,1)$ and $p(1) = (x,y) \in X - \{(0,1)\}$. Define $A := \{ t\in [0,1] \mid p(t) = (0,1)\}$.
    \begin{itemize}
        \item $A$ is closed, since it is the preimage of the closed singleton $\{(0,1)\}$.
        \item $A$ is open: fix $t_0 \in A$. By the continuity of $p$, there exists $\delta >0$ such that $\| p(t) - (0,1)\| < \frac{1}{2}$, $t \in [0,1] \cap (t_0 - \delta ,t_0+ \delta)$. Hence, $p(t)$ cannot lie on the x-axis. Therefore, by the def of $X$, x-coordinate of $p(t)$ must belong to $\{0\}\cup \{1/n \mid n\geq 1\}$. \\
        Let $I:= [0,1] \cap (t_0 - \delta,t_0 + \delta)$. Consider the cont fct $f := \pi_x \circ p: I \rightarrow \mathbb{R}$ where $\pi_x(a,b) =a$ is the x-coordinate. Since $I$ is connected, then $f(I)$ is connected. But $f(I) \subseteq \{0\} \cup \{1/n \mid n \geq 1\}$, whose only connected subsets are singletons. Since $f(t_0)=0$, then we deduce $f(I) =\{0\}$. Hence, $p(t) =(0,1)$ for all $t \in I$, i.e.: $I \subseteq A$, which means $A$ is open.
    \end{itemize}
    Therefore, $A$ is non-empty, closed, and open in the connected interval $[0,1]$, so $A=[0,1]$. Hence, any such path is constant, and no path connects $(0,1)$ to another point of $X$.
    \item $X$ is connected. The proof is omitted here, but it relies on showing that any separation would disconnect the horizontal interval from the vertical combs, which is impossible.
\end{itemize}

\begin{definition}{Concatenation}\\
Let $X$ be a topological space and let $\alpha, \beta: [0,1] \to X$ be two paths with $\alpha(1) = \beta (0)$. Then define the \textit{concatenation} of $\alpha$ and $\beta$, denoted as $\alpha * \beta$, by $\alpha * \beta: [0,1] \to X$ with
$$\alpha * \beta (t) := \begin{cases}
    \alpha (2t) &,if\; \; t\in [0,1/2]\\
    \beta (1-2t) &,if\;\; t\in [1/2,1]
\end{cases}$$
\end{definition}

\begin{definition}{Path-Equivalence}\\
Let $x,y \in X$, topological space. We say $x$ and $y$ are path-equivalent if there exists a path $\gamma : [0,1] \to X$, s.t.:
$$\gamma (0) = x, \gamma (1) = y$$
\end{definition}

\begin{proposition}
    For every topological $X$, path-equivalence defines an equivalence relation.
\end{proposition}

\begin{definition}{Path-components}\\
Let $X$ be a topological space. We call the path-equivalence classes of points the \textit{path-components} of $X$, denoted as $\Pi_0 (X)$.

In particular, if $X$ is non-empty, consisting of a simple path-component, then we write $\Pi_0(X) =0$.
\end{definition}

\begin{proposition}{Path-Component Proposition}\\
Let $X$ be a topological space.
\begin{enumerate}
    \item $X$ is the disjoint union of its path-components.
    \item Each path-component of $X$ is path-connected.
    \item Each path-component $C$ of $X$ is a maximal path-connected subsets, i.e.: if $C \subsetneq D$, then $D$ is NOT path-connected.
    \item Let $A \subseteq X$ be a subset of $X$.
    \begin{enumerate}
        \item If no point in $A$ is path-equivalent to a point in $X \setminus A$, then $A$ is the union of path-components of $X$.
        \item If $A$ is path-connected, then $A$ is contained in a path-component of $X$.
        \item If $A$ is path-connected and if no point in $A$ is path-component to a point in $X \setminus A$, then $A$ is a path-component of $X$.
    \end{enumerate}
\end{enumerate}
\end{proposition}

\begin{proof}
    It's trivial for 1,2,3,4(b),4(c). As for 4(a), suppose on the contrary that there are path-components $\{C_i\}_{i \in I}$, s.t.:
    $$\bigcup_{i \neq j} C_i \subsetneq A \subsetneq \bigcup_{i\in I} C_i$$
    for some $j \in I$. Consider $a \in A \setminus \bigcup_{i \neq j} C_i \subseteq C_j$ and $b \in C_j \setminus A$. Note that $a,b \in C_j$, contradicting the assumption of 4(a).
\end{proof}

\begin{definition}{Connectable}\\
Let $x$ and $y$ be two points in a topological space $X$. We say $x$ and $y$ are connectable if there exists a connected subset of $X$ containing both $x$ and $y$.
\end{definition}

\begin{proposition}
    For every topological $X$, the notion of being connectable gives an equivalence relation.
\end{proposition}

\begin{definition}{Component}\\
Let $X$ be topological. We call the equivalence classes of connectable points on $X$ the \textit{component} of $X$.
\end{definition}

\begin{proposition}{Component Proposition}\\
Let $X$ be a topological space.
\begin{enumerate}
    \item As a set $X$ is the disjoint union of its components.
    \item Each component of $X$ is connected.
    \item Each component $C$ of $X$ is a maximal connected subset of $X$, i.e.: if $C \subsetneq D$, then $D$ is NOT connected.
    \item Let $A \subseteq X$ be a subset of $X$.
    \begin{enumerate}
        \item If $C$ is a component of $X$, then $A \cap C$ is a component of $A$.
        \item If $A$ is both closed and open in $X$, then $A$ is the union of components of $X$.
        \item If $A$ is both closed the open in $X$ and if $A$ is connected, then $A$ is a component of $X$.
    \end{enumerate}
    \item Each component of $X$ is the union of path-components of $X$.
\end{enumerate}
\end{proposition}

\begin{proof}
    It's trivial for 1,2,3,4(a),4(c),5. As for 4(b), let $C$ be a component of $X$. Note that $A \cap C$ is both closed and open in $C$. By (2), the connectivity of $C$, $A \cap C = \emptyset$ or $A \cap C = C$.
\end{proof}

\begin{proposition}
    Let $X$ be a topological space.
    \begin{enumerate}
        \item Every component of $X$ is closed subset of $X$.
        \item If $X$ has only finitely many components, then each component is also an open subset of $X$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}
        \item Let $C$ be a component of $X$. If $C=X$, then we're done, since $\emptyset \in \mathcal{T}_X$. Otherwise, take $P \in X \setminus C$. Then by the component proposition 3, $C \cup \{P\}$ is NOT connected. Therefore, by def there exist open (in $C\cup \{P\}$) nonempty disjoint $U,V$, s.t.:
        $$C \cup \{P\} = U \sqcup V$$
        WLOG, let $C \cap U \neq \emptyset$. Since $C$ is connected and $C= (U \cap C) \sqcup (V \cap C)$, then $V \cap C = \emptyset$. Hence, $V=\{P\}$, open in $C \cup \{P\}$. Therefore, by the subspace topology, there exists an open subset $W$ in $X$, s.t.:
        $$W \cap (C \cup \{P\}) = V = \{P\}$$
        Hence, $W\cap C = \emptyset$ and $P \in W$, which follows that $X \setminus C$ is open $\Leftrightarrow$ $C$ is closed.

        \item Let $C$ be a component of $X$. By assumption, $$X \setminus C = \bigcup_{i=1}^n C_i$$
        where $C_i$'s are components of $X$. Since the union is finite and each $C_i$ is closed by 1, then $X \setminus C$ is closed and hence $C$ is open.
    \end{enumerate}
\end{proof}

\begin{proposition}
    Let $f: X \to Y$ be a continuous map between two topological spaces.
    \begin{enumerate}
        \item Let $C$ be a (path-)component of $Y$. The preimage $f^{-1} (C)$ is the union of (path-)components $X$.
        \item Let $B$ be a (path-)component of $X$. There exists a (path-)component $C$ of $Y$, s.t.: $f (B) \subseteq C$.
    \end{enumerate}
\end{proposition}

\section{Compactness}

\begin{definition}{Cover}\\
Let $X$ be a topological space. An open cover of $X$ is a collection of open sets $U_i$, $i\in I$, s.t.: $X= \bigcup_{i\in I} U_i$. 
Moreover, a subcover of $X$ is a subcollection $\{U_j \mid j \in J \subseteq I\}$ s.t.: $X = \bigcup_{j\in J} U_j$.
\end{definition}

\begin{definition}{Compact}\\
Let $X $ be a topological space. We say $X$ is compact if $\forall$ open cover of $X$, there is always a subcover with finitely many open sets.
\end{definition}

\noindent\textbf{Remark:} A compact set can be regarded as a point.


\noindent\textbf{Example:} 
\begin{itemize}
    \item Heine-Borel Theorem:
    $$X \subseteq \mathbb{R}^n \;\;cpt \Leftrightarrow X \;\; closed \;\; and \;\; bounded.$$
    \item Let $K \subseteq \mathbb{R}^n$ be cpt and let $\mathcal{C}(K) := \{f:K \rightarrow \mathbb{R} \}$. Define a metric: 
    $$d_\infty (f,g) := \sup_{x\in K} \{| f(x) -g(x) | \}.$$
    Then we have Ascoli-Aizela Theorem:
    $$A \subseteq \mathcal{C}(K) \;\; cpt \Leftrightarrow A \;\; closed, \;\; bounded,\;\; and \;\; equicontinuous.$$
\end{itemize}

\noindent\textbf{Remark:}\\
$A \subseteq X$ is cpt if the following two equivalent conditions hold:
\begin{itemize}
    \item $A$ is cpt using subspace topology;
    \item For any $\{U_i\}_{i\in I}$ open in $X$, s.t.: $A \subseteq \bigcup _{i\in I} U_i$, there is a finite subcover $A \in \bigcup_{j=1}^n U_{i_j}$.
\end{itemize}

\begin{proposition}
    Let $f: X\rightarrow Y$ be a cont fct between two topological spaces. If $A \subseteq X$ is cpt, then $f(A)$ is cpt.
\end{proposition}

\begin{proof}
    Suppose $f(A) \subseteq \bigcup_{i \in I} U_i$ open cover in $Y$. Then $A = f^{-1}(\bigcup_{i \in I} U_i) = \bigcup_{i \in I} f^{-1}(U_i)$.
    By the cptness of $A$, $A \subseteq \bigcup_{j=1}^n f^{-1}(U_{i_j})$. Hence, $f(A) \subseteq \bigcup_{j=1}^n U_{i_j}$.
\end{proof}

\begin{corollary}
    Let $K$ be cpt and let $f :K \rightarrow \mathbb{R}$ be cont. Then $f(K)$ is closed and bounded in $\mathbb{R}$. In particular, there are $m,M \in \mathbb{R}$, s.t.: $m \leq f(x) \leq M$ for all $x \in K$.
\end{corollary}

\begin{lemma}
    Let $X$ be Hausdorff and let $K \subseteq X$ be cpt. Then for ant $x \in X-K$, there are two open sets $U \ni x$ and $V \supseteq K$, s.t.: $U \cap V =\emptyset$.
\end{lemma}

\begin{proof}
    For any $k \in K$, clearly $x \neq k$, so we have $U_k \ni x$ and $V_k \ni k$ such that $U_k \cap V_k = \emptyset$, by the Hausdorff-ness of $X$. Therefore, $\{ V_k \mid k \in K\}$ is an open cover of $X$, i.e.: $\bigcup_{i=1}^n V_{k_i} \supseteq K$. Now take $U := \bigcap _{i=1}^n U_{k_i}$ and $V:= \bigcup_{i=1}^n V_{k_i}$. Then we're done.
\end{proof}

\begin{corollary}
    Let $X$ be Hausdorff and let $K \subseteq X$ be cpt. Then $K$ is closed.
\end{corollary}

\begin{proof}
    For any $x \in X-K$, by lemma 4.25, there exists an open set $U_x \ni x$ s.t.: $U_x \subseteq X-K$. Then $X-K$ is open.
\end{proof}

\begin{proposition}
    If $X,Y$ are cpt, then $X \times Y$ is cpt using the product topology.
\end{proposition}

\begin{proof}
    Suppose $A = \{W_i \} _{i\in I}$ is an open cover of $X\times Y$. Then $X\times Y = \bigcup_{i\in I} W_i$. We can write: 
    $$X\times Y = \bigcup_{i\in I} W_i = \bigcup_{i\in I} \bigcup_{j\in J} U_{ij} \times V_{ij}$$
    where $U_{ij} \in \mathcal{T}_X$ and $V_{ij} \in \mathcal{T}_Y$.

    For each $y\in Y$, $X \times \{y\} \subseteq \bigcup_{i,j} U_{ij} \times V_{ij} =: \bigcup_{k} U_{k} \times V_{k}$ where $\{ k=(i,j) \mid i \in I, j \in J\}$. Since $X\times \{y\}$ is cpt, then $X \times \{y\} \subseteq \bigcup_{s \in F_y} U_{s} \times V_{s}$ where $|F_y| < \infty$. Then:
    \begin{align*}
        y \in \bigcap_{s \in F_y} V_s =: V_y &\Rightarrow Y= \bigcup_{y\in Y} V_y \\
        &\overset{Y \; cpt}{\Rightarrow} Y= \bigcup _{l=1}^m V_{y_l}
    \end{align*}
    Hence,
    $$X\times Y = \bigcup_{l=1}^m X\times V_{y_l} \subseteq \bigcup_{l=1}^m \bigcup_{s \in F_{}y_l} U_{s} \times V_{s} \subseteq \bigcup_{l=1}^m \bigcup_{s \in F_{}y_l} W_{s}. $$
\end{proof}

\begin{definition}{Product Topology for infinite product}\\
Let $\{(X_i, \mathcal{T}_i)\}_{i\in I}$ topological spaces. Then the product topology of $\Pi_{i\in I} X_i := \{(x_i)_{i\in I} \mid x_i \in X_i\}$ has a basis of the form:

$\{ \Pi_{i\in I} U_i \mid U_i \in \mathcal{T}_i$ for all $i\in I$ and $U_i=X_i$ for all but finitely many $i\in I\}$.\\
Indeed, the product topology defined above is the coarsest topology satisfying the property that

$p_i :\Pi_{j\in I} X_j \rightarrow X_i$ with $p_i(x_j) := x_i$ is cont for all $i\in I$.
\end{definition}

\begin{lemma}
    Let $\mathcal{B}$ be a collection of open subsets of $X \times Y \times Z$ with $Y$ compact. Suppose that there exists a point $x_0 \in X$, s.t.: for any open neighborhood of $x_0$ $U$, $U \times Y \times Z$ is NOT finitely covered by $\mathcal{B}$. Then $\exists y_0 \in Y$, s.t.: $\forall \;open\;U\ni x_0,V\ni y_0$, $U \times V \times Z$ is NOT finitely covered by $\mathcal{B}$.
\end{lemma}

\begin{proof}
    Suppose on the contrary that for any $y \in Y$, there exists an open $U_y \ni x_0,V_y \ni y$, s.t.:
    $$U_y \times V_y \times Z \subseteq \bigcup_{j=1}^{m_y} W_{y,j}$$
    where $W_{y,j} \in \mathcal{B}$.

    Since $\{ V_y \mid y \in Y \}$ is an open cover of cpt $Y$, then there is an open finite subcover of $Y$ $\{ V_{y_{i}} \}_{i=1}^n$.
    Then set $U := \bigcap_{i=1}^n U_{y_i}$. Then:
    $$U \times Y \times Z \subseteq \bigcup_{i=1}^n (U_{y_i} \times V_{y_i} \times Z) \subseteq \bigcup_{i=1}^n \bigcup_{j=1}^{m_{y_i}} W_{y_i,j}$$
    contradicting the hypothesis of lemma.
\end{proof}

\begin{theorem}{Tychonoff's Theorem}\\
Let $\{X_i \mid i\in I \}$ be cpt topological spaces. Then $\Pi_{i\in I} X_i$ is cpt in product topology.
\end{theorem}

\noindent\textbf{Remark:} We'll only prove the case when $I$ is countable. For the case when $I$ is uncountable, we need axiom of choice (well-ordering).

\begin{proof}
    Let $\mathcal{B}$ be a collection of open sets of $X = \Pi_{i=1}^\infty X_i$, s.t.: $\mathcal{B}$ does NOT cover $X$ finitely.
    \begin{itemize}
        \item \textbf{Claim 1}: there exists a point $x_1 \in X_1$, s.t.: for any $U_1 \ni x_1$ open in $\mathcal{B}$, $\mathcal{B}$ does NOT cover $U_1 \times X_2 \times X_3 \times \cdots$ finitely.

        \textit{Proof of claim 1}: Suppose on the contrary that for any $x \in X_1$, there exists $U_x \ni x$ open, s.t.:
        $$U_x \times X_2 \times X_3 \times \cdots \subseteq \bigcup_{i=1}^{m_x} W_{x,i}.$$
        Since $\{ U_x \}$ covers $X_1$, then we have $\{ U_{x_j}\}_{j=1}^{p}$ covers $X_1$. Hence,
        $$X_1 \times X_2 \times \cdots \subseteq \bigcup_{j=1}^p U_{x_j} \times X_2 \times X_3 \times \cdots \subseteq \bigcup_{j=1}^p \bigcup_{i=1}^{m_{x_j}} W_{x_j ,i}.$$
        Therefore, $\mathcal{B}$ covers $X_1 \times X_2 \times X_3 \times \cdots$ finitely, contradiction!
    \end{itemize}
    Then by claim 1 and lemma 4.29, if we regard $U_1$ in claim 1 as $X$ in lemma 3.28, $X_2$ as $Y$ in lemma 4.29, and $X_3 \times X_4 \times \cdots$ as $Z$ in lemma 3.28, then there exists an point $x_2 \in X_2$, s.t.: for any open $U_1, \ni x_1,U_2,\ni x_2$, $\mathcal{B}$ does NOT cover $U_1 \times U_2 \times X_3 \times X_4 \times \cdots$ finitely.

    Repeat this procedure. Then we get (*): $\exists \mathbf{x} = (x_1, \cdots, x_n,\cdots) \in \Pi_{i=1}^\infty X_i$, s.t.: for any $n \in \mathbb{N}_+$, $\forall U_i \ni x_i,(i=1,\cdots n)$, $U_1 \times \cdots \times U_n \times X_{n+1} \times \cdots$ can NOt be covered by $\mathcal{B}$ finitely.
    \begin{itemize}
        \item \textbf{Claim 2}: $\mathbf{x} \notin \bigcup_{W \in \mathcal{B}}W$.

        \textit{Proof of claim 2}: Suppose on the contrary that $\mathbf{x} \in W_b$ for some $W_b \in \mathcal{B}$. Then $\mathbf{x} \in W_b \subseteq \bigcup (\Pi_{i=1}^\infty U_i)$, where $(\Pi_{i=1}^\infty U_i)$ is in the basis of product topology of $X$. Hence, 
        $$\mathbf{x} \in \Pi_{i=1}^\infty U_i \Rightarrow \mathbf{x} \in U_1 \times \cdots U_N \times X_{N+1} \times \cdots \subseteq W_b$$
        contradicting (*).
    \end{itemize}
    Then by claim 2, we implies that $X$ is NOT contained in $\bigcup_{W \in \mathcal{B}} W$.
\end{proof}

\begin{lemma}
    Any closed subset of a compact set is compact.
\end{lemma}

\begin{proof}
    Suppose $X$ is cpt, $V$ is closed in $X$, and $\{ U_i \}_{i\in I}$ is an open cover of $V$. Then $\{ U_i \}_{i\in I} \cup (X-V)$ is an open cover of $X$. Hence, by the cpt-ness of $X$, $\{ U_{i_k} \} _{k=1} ^m \cup (X-V)$ covers $X$. Therefore, $\{ U_{i_k} \} _{k=1} ^m $ covers $V$.
\end{proof}

\begin{theorem}
    Suppose $X$ is compact, $Y$ is Hausdorff, and $f: X\rightarrow Y$ is cont and bijective. Then $f^{-1}$ is cont, and moreover, $f$ is a homeomorphism.
\end{theorem}

\begin{proof}
    Take a closed set $V \subseteq X$, and by lemma 3.30, $V$ is cpt. Then by the continuity of $f$, $f(V)$ is cpt. Since $Y$ is Hausdorff, then $f(V)$ is closed. Therefore, $(f^{-1})^{-1}(V) = f(V)$ is closed whenever $V$ is closed, i.e.: $f^{-1}$ is cont.
\end{proof}

\noindent\textbf{Example:} Consider $f : S^1 \times S^1 \rightarrow \mathbb{R}^3$ with $f(\theta , \phi ) : = \{ (R + r\cos \theta) \cos \phi , (R + r\cos \theta) \sin \phi , r \sin \phi \}$, where $S ^1 := \{ (\cos\theta , \sin \theta) \mid \theta \in \mathbb{R} \}$. Then $f(S^1 \times S^1)$ forms a torus, which can be denoted by $T^2$.\\
It's easy to check that $f$ is cont, $ S^1 \times S^1$ is cpt (i.e.: closed and bdd in $\mathbb{R} ^2$), and $f( S^1 \times S^1)$ is Hausdorff. Then by theorem 3.30, $ S^1 \times S^1 \cong  f(S^1 \times S^1) = T^2$ are homeomorphic.

\begin{definition}{Cantor Set}\\
Let $C_0 := [0,1]$. Then we define $\{ C_i\}_{i=1}^\infty$ iteratively as follows:
$$C_i := \{ \frac{1}{3} x \mid x \in C_{i-1} \} \cup \{ \frac{2}{3} + \frac{1}{3} x \mid x \in C_{i-1} \} (= \frac{1}{3}C_{i-1} \cup (\frac{2}{3} + \frac{1}{3}C_{i-1}))$$
Then we define \textit{Cantor Set} by
$$C:= \bigcap _{i=0}^\infty C_{i}$$
\end{definition}

\begin{proposition}
    The Cantor Set is compact in $\mathbb{R}$.
\end{proposition}

\section{Normal Topological Space}

\begin{definition}{Normal Topological Space}\\
A topological space $X$ is called \textit{normal} if any two disjoint closed subsets $A,B$ of $X$ can be separated by open neighborhoods, i.e.: there exists two open sets $U,V$, s.t.: $A \subseteq U, B \subseteq V$ and $U \cap V =\emptyset$.
\end{definition}

\begin{proposition}
    Every compact Hausdorff space is normal.
\end{proposition}

\begin{proposition}{Metric Spaces-Normal Proposition}\\
Let $(X,d)$ be a metric space. Then $(X,\mathcal{T}_{d})$ is a normal topological space.
\end{proposition}

\begin{proof}
    Given two disjoint closed subsets $A,B$ of $B$, if one of $A$ or $B$ is empty, then the desired open neighborhoods are $\emptyset$ and $X$. Otherwise, $A,B$ are both non-empty. Note that $\forall x \in X$,
    $$d(x,A)+ d(x,B) >0$$
    So we can consider a well-defined map $f:X \to \mathbb{R}$ given by
    $$f(x) := \frac{d(x,A)}{d(x,A) + d(x,B)}$$
    which is continuous by the Distance-to-Subspace Proposition.1. It's clear that $f|_A \equiv 0, f|_B \equiv 1$. Then
    $$A \subseteq U:= f^{-1}((-\infty, \frac{1}{4})), B\subseteq V:= f^{-1}((\frac{3}{4}, \infty))$$
    are the two desired open neighborhoods.
\end{proof}

\begin{proposition}{Normal-Open-Closed-Neighborhood Proposition}\\
Let $X$ be a topological space and let $A \subseteq X$ be a closed subset and $U \subseteq X$ be an open subset with $A \subseteq U$. If $X$ is normal, then there exists an open $V \subseteq X$ in $X$, s.t.:
$$A \subseteq V \subseteq \overline{V} \subseteq U$$
\end{proposition}

\begin{proof}
    Since $A \subseteq U$, then $A \cap (X \setminus U)= \emptyset$, where $A,X\setminus U$ are closed. By the normality of $X$, there are disjoint open neighborhoods $V$ of $A$, $W$ of $X\setminus U$. It suffices to show that $\overline{V} \subseteq U$. Note that $V \subseteq X \setminus W$ with $X \setminus W$ closed. Hence,
    $$\overline{V} \subseteq \overline{X \setminus W} = X \setminus W\subseteq U$$
\end{proof}

\begin{proposition}{Closed Disjoint Embedding Proposition}\\
Let $Y$ be a normal space.
\begin{enumerate}
    \item Let $A_1,\cdots, A_k \subseteq Y$ be distinct subsets. If $A_i$'s are closed in $Y$, then the natural map $A_1 \sqcup \cdots \sqcup A_k \to A_1 \cup \cdots \cup A_k$ is a homeomorphism.

    \item Let $X_1,\cdots, X_k$ be compact spaces and let $\varphi_i: X_i \to Y$ be embeddings with disjoint images. If $Y$ is Hausdorff, then the map
    $$\bigsqcup_{i=1}^k \varphi_i : \bigsqcup_{i=1}^k X_i \to Y$$
    is an embedding.
\end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}
        \item Given $j \in \{1,\cdots,k\}$, note that $\bigcup_{i \neq j} A_i$ is closed and that it is disjoint with $A_j$ closed in $Y$. Then by the normality of $Y$, there exists an open neighborhood of $A_j$ that doesn't intersect any of the other $A_i$. Hence, by the Disjoint Embedding Proposition, we're done.

        \item By the compact-ness of $X_i$ and the Hausdorff-ness of $Y$, the images of embeddings $\varphi_i(X_i)$'s are closed in $Y$. Then by 1, we're done.
    \end{enumerate}
\end{proof}

\section{Urysohn's Lemma}

\begin{lemma}{Urysohn's Lemma}\\
Let $X$ be a topological space. The followings are equivalent:
\begin{enumerate}
    \item For any two disjoint closed subsets $A,B$, any $r<s \in \mathbb{R}$, there exists a continuous map $f: X \to [r,s]$ with $f|_A \equiv r, f|_B \equiv s$.
    \item $X$ is normal.
\end{enumerate}
\end{lemma}

\begin{proof}
    \begin{itemize}
        \item (1$\Rightarrow$2): For any disjoint closed $A,B$ in $X$, by 1, $f^{-1}((-\infty, \frac{r+s}{2}))$ and $f^{-1}((\frac{r+s}{2},\infty))$ are the desired disjoint open neighborhoods of $A$ respectively $B$.
        \item (2$\Rightarrow$1): Given two disjoint closed $A,B$ subsets of $X$, it suffices to show that there is a continuous map $f:X \to [0,1]$, s.t.: $f|_A\equiv 0,f|_B \equiv 1$.
        \begin{itemize}
            \item \textbf{Claim 1:} Set $P:= [0,1] \cap \mathbb{Q}$. There exists a fammily $\{U_s\}_{s \in P}$ of open subsets of $X$ with the following properties:
            \begin{enumerate}
                \item $A \subseteq U_0$
                \item $\forall p\in P$ with $p<1$, we have $U_p \subseteq X \setminus B$
                \item $\forall s,t\in P$ with $s<t$, we have $\overline{U}_s \subseteq U_t$.
            \end{enumerate}

            \item \begin{proof}
                \textit{of claim 1:} Since $P$ is countable, then there is a bijective $\varphi: \mathbb{N} \cup \{0\} \to P$. WLOG, let $\varphi(0)=0$ and $\varphi (1)=1$. Set $U_1:= X\setminus B$, which is open by the closed-ness of $B$. Since $X$ is normal and since $A$ is closed, then by the Normal-Open-Closed-Neighborhood Proposition, there is an open neighborhood $U_1$ of $A$, s.t.: $\overline{U}_0 \subseteq U_1$.\\
                Now we suppose that we're already constructed $U_{\varphi(0)}= U_0, U_{\varphi(1)}= U_1 ,U_{\varphi(2)} ,\cdots, U_{\varphi(n)}$ with the required properties. Then set $P_n:= \{ \varphi(0) ,\cdots, \varphi (n)\}$ and $y:= \varphi(n+1)$. By the injectivity of $\varphi$, since $\varphi(0)=0$ and $\varphi(1)=1$, then $y \in (0,1)$. Since $P_n$ is finite, then $\exists x,z \in P_n$, s.t.:
                $$p\leq x<y,\forall p\in P_n \;\;and\;\; y< z\leq q ,\forall q\in P_n$$
                Then by the property 3, $\overline{U}_x \subseteq U_z$. Hence, by the Normal-Open-Closed-Neighborhood Proposition, there is an open $U_y$, s.t.:
                $$\overline{U}_x \subseteq U_y \subseteq \overline{U}_y \subseteq U_z$$
                Therefore, we're done.
            \end{proof}
        \end{itemize}

        $\forall p \in (-\infty,0) \cap \mathbb{Q}$, we set $U_p := \emptyset$ and $\forall p\in (1,\infty) \cap \mathbb{Q}$, we set $U_p :=X$. Consider $f: X \to \mathbb{R}$ given by
        $$f(x) := \inf \{p \in \mathbb{Q} \mid x\in U_p\}$$
        By the properties 1,2, $f|_A \equiv 0,f|_B \equiv 1$ and $f(X) \subseteq [0,1]$. Then it remains to prove the continuity of $f$.

        \begin{itemize}
            \item \textbf{Claim 2:} Let $r \in \mathbb{Q}$.
            \begin{enumerate}
                \item $x \in \overline{U}_r \Rightarrow f(x) \leq r$, i.e.: $f(x) >r \Rightarrow x \notin \overline{U}_r$.
                \item $x \notin U_r \Rightarrow f(x) \geq r$, i.e.: $f(x) <r \Rightarrow x \in U_r$.
            \end{enumerate}

            \item \begin{proof}
                \textit{of claim 2:} It's trivial for 1,2 in claim 2 if $r<0$ of $r >1$. Now we suppose that $r \in P= [0,1] \cap \mathbb{Q}$. Then by the property 3 in claim 1, 
                $$x \in \overline{U}_r \overset{3}{\Rightarrow} x \in U_s, \forall s \in P,with \;s>r \overset{P\;dense\;in\;[0,1]}{\Longrightarrow} f(x) \leq r$$
                and similarly for 2.
            \end{proof}

            \item \textbf{Claim 3:} Given any $r \in \mathbb{Q}$, we have
            $$f^{-1} ([0,r]) = \bigcap_{n \in \mathbb{N}_+} \overline{U}_{r+ \frac{1}{n}}, f^{-1}([0,r)) = \bigcup_{n \in \mathbb{N}_+} U_{r-\frac{1}{n}}$$
            \item \begin{proof}
                \textit{of claim 3:}
                \begin{itemize}
                    \item $f^{-1}([0,r]) = f^{-1} (\bigcap_{n \in \mathbb{N}_+} [0, r+\frac{1}{n})) = \bigcap_{n \in \mathbb{N}_+} f^{-1} ([0,r+\frac{1}{n})) \overset{claim.2.2}{\subseteq} \bigcap_{n \in \mathbb{N}_+}  U_{r+\frac{1}{n}} \subseteq \bigcap_{n \in \mathbb{N}_+}  \overline{U}_{r+\frac{1}{n}} \overset{claim.2.1}{\subseteq} \bigcap_{n \in \mathbb{N}_+} f^{-1}([0, r+\frac{1}{n}]) = f^{-1} (\bigcap_{n \in \mathbb{N}_+} [0,r+\frac{1}{n}]) =f^{-1}([0,r])$.

                    \item $f^{-1}([0,r)) = f ^{-1} (\bigcup_{n \in \mathbb{N}_+} [0,r- \frac{1}{n})) = \bigcup_{n \in \mathbb{N}_+} f^{-1} ([0,r- \frac{1}{n})) \overset{claim.2.2}{\subseteq} \bigcup_{n \in \mathbb{N}_+} U_{r- \frac{1}{n}} \subseteq \bigcup_{n \in \mathbb{N}_+} \overline{U}_{r- \frac{1}{n}} \overset{claim.2.1}{\subseteq} \bigcup_{n \in \mathbb{N}_+} f^{-1} ([0,r -\frac{1}{n}]) = f^{-1}(\bigcup_{n \in \mathbb{N}_+} [0,r- \frac{1}{n}]) = f^{-1} ([0,r))$.
                \end{itemize}
            \end{proof}

            \item \textbf{Claim 4:} $f$ is continuous.
            \item \begin{proof}
                \textit{of claim 4:} $\forall c<d \in \mathbb{Q}$,
                $$f^{-1} ((c,d)) = f^{-1} ([0,d)) \setminus f^{-1} ([0,c]) \overset{claim.3}{=} \bigcup_{n \in \mathbb{N}_+} U_{d-\frac{1}{n}} \setminus \bigcap_{n \in \mathbb{N}_+} \overline{U}_{c+\frac{1}{n}}$$
                is open in $X$.
            \end{proof}
        \end{itemize}
    \end{itemize}
\end{proof}

\section{Tietze Extension Theorem}

\begin{theorem}{Tietze Extension Theorem}\\
Let $X$ be a topological space, let $A \subseteq X$ be a closed subset of $X$ and let $f: A \to \mathbb{R}$ be a continuous map. If $X$ is normal, then $\exists$ a continuous map $g: X \to \mathbb{R}$ with $g|_A=f$. If $A$ is non-empty, then one can find such $g$, s.t.:
$$\inf g(X)=\inf f(A) \in \mathbb{R} \cup \{-\infty\}, \sup g(X)= \sup f(A) \in \mathbb{R} \cup \{\infty \}$$
\end{theorem}

\begin{theorem}{Tietze Extension Theorem $\mathrm{II}$}\\
Let $X$ be a topological space and let $A \subseteq X$ be a closed subset of $X$. Suppose that for $i= 1,\cdots, n$, we're given a closed interval $I_i \subseteq \mathbb{R}$ and suppose that we're given a continuous map $f: A \to I_1 \times \cdots \times I_n \subseteq \mathbb{R}^n$. If $X$ is normal, then there is a map $g: X \to I_1 \times \cdots \times I_n$ with $g|_A =f$.
\end{theorem}



\section{Local and Regional Properties}

\begin{definition}
    Let $P$ be a property of topological spaces and let $X$ be a topological space.
    \begin{itemize}
        \item We say $X$ is \textit{regionally $P$} if for any $x \in X$, any $U \ni x$, there exists $V \subseteq X$ with property $P$, s.t.:
        $$x \in V \subseteq U$$
        \item We say $X$ is \textit{locally $P$} if for any $x \in X$, any $U \ni x$, there exists \textbf{open} $V \subseteq X$ with property $P$, s.t.:
        $$x \in V \subseteq U$$
    \end{itemize}
\end{definition}

\noindent\textbf{Remark:} Locally property $\Rightarrow$ regionally property.

\begin{proposition}{Basis-with-property Proposition}\\
Let $P$ be a property of topological spaces and let $X$ be a topological space. If $X$ is locally $P$, then the open subsets of $X$ that have $P$ form a basis for the topology of $X$.
\end{proposition}

\begin{proposition}{Locally Path-Connected Proposition}\\
Let $X$ be a topological space.
\begin{enumerate}
    \item If $X$ is locally path-connected, then the following statements hold:
    \begin{enumerate}
        \item $X$ is connected $\Leftrightarrow$ $X$ is path-connected.
        \item The components of $X$ are precisely the path-components of $X$.
    \end{enumerate}
    \item $X$ is locally path-connected, then it is also locally connected.
    \item If $X$ is locally connected, then every component of $X$ is both closed and open.
\end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}
        \item \begin{enumerate}
            \item ($\Leftarrow$): Obviously. ($\Rightarrow$) is a special case of $(b)$.
            \item \begin{itemize}
                \item \textbf{Claim 1:} Each path-component of $X$ is open.
                \begin{proof}
                    Let $P$ be a path-component of $X$. Pick $x \in X$. For any $y \in P$, since $X$ is path-connected, then we can take a path $\gamma : [0,1] \to X$ given by $\gamma (0) = x$ and $\gamma (1) =y$. Since $X$ is locally path-connected, then there exists an open path-connected $U$, s.t.: $y \in U \subseteq P$. Hence, $\gamma ([0,1]) \cup U$ is path-connected. Therefore, by the path-component proposition 4(b), $\gamma ([0,1]) \cup U \subseteq P$. Then we have:
                    $$y \in U \subseteq \gamma ([0,1]) \cup U \subseteq P$$
                    with $U$ open which follows that $P$ is open.
                \end{proof}
                \item \textbf{Claim 2:} Each path-component of $X$ is closed.
                \begin{proof}
                    Let $P$ be a path-component of $X$. By the path-component proposition 1, $X \setminus P = \bigsqcup_i P_i$, where $P_i$'s are the components of $X$. By claim 1, $X \setminus P$ is open and hence $P$ is closed.
                \end{proof}
            \end{itemize}
            Let $C$ be a component of $X$. By the component proposition 2,5, $C$ is connected and there exists a family of path-components $\{P_i \}_{i\in I}$, s.t.:
            $$C= \bigsqcup_{i \in I} P_i$$
            By claim 1 and claim 2, each $P_i$ is both open and closed. Since $C$ is connected, then $I$ consists of a single element, i.e.: $C$ is path-component of $X$.
        \end{enumerate}
        \item Obviously.
        \item Let $C$ be a component of $X$. Then by proposition 4.29.1, $C$ is closed. For any $x \in C$, since $X$ is locally connected, then there exists an open connected $U$, s.t.: $x \in U \subseteq C$. Therefore, $C$ is open.
    \end{enumerate}
\end{proof}

\begin{proposition}
    Let $X$ be a Hausdorff topological space.
    \begin{enumerate}
        \item If $X$ is compact, then $X$ is also regionally compact.
        \item If $\forall x \in X$, there exists a compact neighborhood of $x$, then $X$ is regionally compact.
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}
        \item Let $x \in X$ and let $U$ be a neighborhood of $x$.

        \noindent\textbf{Claim:} There exists an open neighborhood $A$ of $x$ with $\overline{A} \subseteq U$.
        \begin{proof}
            Since $X$ is Hausdorff, then $\{x\}$ is closed in $X$. By proposition 4.46, $X$ is normal. Hence, we can separate the closed sets $\{x\}$ and $X \setminus U^\circ$ by open disjoint neighborhoods $A$ and $B$, i.e.: $A \cap B =\emptyset$, $x \in A$ and $X \setminus U^\circ \subseteq B \Rightarrow X \setminus B \subseteq U^\circ \subseteq U$. Hence, 
            $$x \in A \subseteq X \setminus B \subseteq U$$
            Since $B$ is open, then $X \setminus B$ is closed and hence 
            $$x \in A \subseteq \overline{A} \subseteq X \setminus B \subseteq U$$
        \end{proof}
        Since $\overline{A}$ is closed and $X$ is compact, then $\overline{A}$ is compact.

        \item For any $x \in X$, any $U \ni x$, by hypothesis, there exists a compact $K \ni x$. By 1, $K$ is regionally compact. Hence, there exists a compact $L$, s.t.: 
        $$x \in L \subseteq U \cap K \subseteq U$$
    \end{enumerate}
\end{proof}

\begin{definition}{Precompact}\\
Let $X$ be a topological space. A subset $A \subseteq X$ is called \textit{precompact} if $\overline{A}$ is compact.
\end{definition}

\begin{proposition}{Regionally Compact–Locally Precompact Proposition}\\
Let $X$ be a Hausdorff space that is regionally compact. Then for any $x \in X$, any open $U \ni x$, there exists an open $V \ni x$, s.t.: $\overline{V}$ is compact and s.t.: $x \in V \subseteq \overline{V} \subseteq U$, i.e.: $X$ is locally precompact.
\end{proposition}

\begin{definition}
    Let $P$ be a property of maps between two topological spaces and let $f : X \to Y$ be a map between two topological spaces.
    \begin{itemize}
        \item We say $f$ is \textit{locally $P$ at $x \in X$} if there exist an open $U \ni x$ and an open $V \ni f(x)$, s.t.: $f: U \to V$ has the property $P$.
        \item We say $f$ is \textit{locally $P$} if it is locally $P$ at any $x \in X$.
    \end{itemize}
\end{definition}

\begin{proposition}
    If a map $f: X \to Y$ between two topological spaces is bijective and a local homeomorphism, then it is a homeomorphism.
\end{proposition}

\begin{proposition}
    Let $f: X \to Y$ be a continuous map. If $Y$ is Hausdorff and regionally compact, then
    \begin{enumerate}
        \item $f$ proper $\Rightarrow$ $f$ closed.
        \item $f$ injective and proper $\Leftrightarrow$ closed embedding.
    \end{enumerate}
\end{proposition}

\begin{proof}
    \begin{enumerate}
        \item Let $A \subseteq X$ be a closed subset of $X$. For any $y \in Y \setminus f(A)$, since $Y$ is regionally compact and Hausdorff, then by proposition 4.52, there exists an open $U \ni y$, s.t.: $y \in U \subseteq \overline{U}$ with $\overline{U}$ compact. Since $f$ is proper, then $f^{-1} (\overline{U})$ is compact and hence so is $f^{-1} (\overline{U}) \cap A$. Therefore, $f(A) \cap \overline{U} = f(f^{-1} (\overline{U}) \cap A)$ is compact. Since $Y$ is Hausdorff, then $f(A) \cap \overline{U}$ is closed in $Y$. Set $V:= U \setminus (f(A) \cap \overline{U}) = U \setminus f(A)$ is an open neighborhood of $y \in Y \setminus f(A)$ that is contained in $Y \setminus f(A)$. Therefore, $Y \setminus f(A)$ is open, i.e.: $f(A)$ is closed.

        \item \begin{itemize}
            \item ($\Rightarrow$): By a, we already then get $f$ is closed. Furthermore, $f$ is closed $\Rightarrow$ $f(X)$ is closed in $Y$ $\Rightarrow$ $f(X)$ is compact in $Y$ $\Rightarrow$ $X$ is compact. Then by theorem 4.42, $f$ is a homeomorphism between $X \cong f(X)$.
            \item ($\Leftarrow$): It suffices to show that $f$ is proper. Let $K \subseteq Y$ be a compact subset of $Y$. Since $f$ is closed, then $f(X)$ is closed $\Rightarrow$ $f(X) \cap K$ is compact. Since $f$ is embedding, then $f^{-1} (f(X) \cap K) = X \cap f^{-1} (K) = f^{-1} (K)$ is compact.
        \end{itemize}
    \end{enumerate}
\end{proof}

\begin{proposition}{Closed-Embedding-to-$\mathbb{R}^n$ Proposition}\\
Let $X$ be a topological space, $n \in \mathbb{N}_+$ and let $f: X \to \mathbb{R}^n$ be an injective continuous map. Suppose that $X$ is the union of compact subsets $\{K_i\}_{i\in I}$, s.t.: for any $r>0$, there are finitely many $i_1, \cdots, i_l \in I$, s.t.:
$$f(K_j) \cap \overline{B}_r^n \neq \emptyset$$
Then $f$ is a closed embedding.
\end{proposition}

\begin{proof}
    \begin{lemma}{Proper Criterion}\\
    Let $f:X \to Y$ be a continuous map between two topological spaces with $Y$ Hausdorff.
    \begin{enumerate}
        \item Let $\{ L_j\}_{j \in J}$ be a family of compact subsets of $Y$ satisfying that each compact subset of $Y$ is contained in some $L_j$. If each $f^{-1} (L_j)$ is compact, then $f$ is proper.

        \item Let $X$ be covered by compact subsets $\{K_i\}_{i \in I}$ and let $L \subseteq Y$ be compact in $Y$. If there are finitely many $i \in I$, s.t.: $f(K_i) \subseteq L$, then $f^{-1} (L)$ is compact in $X$.
    \end{enumerate}
    \end{lemma}

    \begin{proof}
        \begin{enumerate}
            \item For any compact $L$ in $Y$, there exists $j \in J$, s.t.: $L \subseteq L_j$. Hence,
            $$f^{-1} (L) = f^{-1} (L \cap L_j) = f^{-1}(L) \cap f^{-1} (L_j)$$
            is compact in $X$ since $f^{-1}(L)$ is closed and $f^{-1} (L_j)$ is compact.
            \item $$f^{-1} (L) = f^{-1} (L) \cap \bigcup_{i\in I,f(K_i \subseteq L)} K_{i}$$
            is compact since $f^{-1} (L)$ is closed and $\bigcup_{i\in I,f(K_i \subseteq L)} K_{i}$ is compact.
        \end{enumerate}
    \end{proof}
    Since $\mathbb{R}^n$ is Hausdorff, then by lemma above (proper criterion)2, for any $r>0$, $f^{-1} (\overline{B}_r^n)$ is compact in $X$. Hence, by lemma above (proper criterion)1, $f$ is proper. Since $\mathbb{R}^n$ is Hausdorff and regionally compact, then proposition 4.55, $f$ is injective and proper $\Leftrightarrow$ $f$ is a closed embedding.
\end{proof}
